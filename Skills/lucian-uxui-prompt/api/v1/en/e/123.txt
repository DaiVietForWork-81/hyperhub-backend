<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];const modes=[['normal','Normal'],['multiply','Multiply'],['screen','Screen'],['overlay','Overlay'],['difference','Difference'],['color','Color']];let i=0;function paint(){root.style.setProperty('--blend',modes[i][0]);q('.mode').textContent=modes[i][1]+' ↻';q('.mode').setAttribute('aria-label','Blend mode '+modes[i][1]+', press for the next mode')}q('.mode').onclick=()=>{i=(i+1)%modes.length;paint()};q('.mode').onkeydown=e=>{if(e.key==='Escape'){i=0;paint()}};})();
</script>
