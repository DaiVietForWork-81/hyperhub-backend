<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var its = Array.prototype.slice.call(root.querySelectorAll('.u066-it'));
  var bs = its.map(function (it) { return it.querySelector('.u066-b'); });
  var msg = root.querySelector('.u066-msg'), back = root.querySelector('.u066-back'), mail = root.querySelector('.u066-mail');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var busy = false, t = 0;
  bs.forEach(function (b) { b.setAttribute('data-l', b.querySelector('.u066-tx').innerHTML); });
  bs.forEach(function (b, i) {
    b.addEventListener('click', function () {
      if (busy || b.classList.contains('ok')) return;
      busy = true;
      root.classList.add('is-pick');
      its.forEach(function (it, k) { if (k !== i) { it.classList.add('gone'); it.querySelector('button').tabIndex = -1; } });
      b.classList.add('busy');
      b.setAttribute('aria-busy', 'true');
      msg.textContent = 'Opening ' + b.getAttribute('data-p') + '…';
      mail.tabIndex = -1;
      t = setTimeout(function () {
        if (!root.isConnected) return;
        busy = false;
        b.classList.remove('busy'); b.classList.add('ok');
        b.removeAttribute('aria-busy');
        b.querySelector('.u066-tx').innerHTML = 'Continue as Seoyeon Kim<small>' + b.getAttribute('data-a') + '</small>';
        msg.textContent = b.getAttribute('data-p') + ' account connected';
        root.classList.add('is-done');
        back.tabIndex = 0;
      }, rm ? 300 : 1500);
    });
  });
  back.addEventListener('click', function () {
    clearTimeout(t); busy = false;
    root.classList.remove('is-pick', 'is-done');
    its.forEach(function (it, k) { it.classList.remove('gone'); it.style.setProperty('--i', k); });
    bs.forEach(function (b) { b.classList.remove('busy', 'ok'); b.tabIndex = 0; b.querySelector('.u066-tx').innerHTML = b.getAttribute('data-l'); });
    mail.tabIndex = 0; back.tabIndex = -1; msg.textContent = '';
    bs[0].focus({ preventScroll: true });
  });
  mail.addEventListener('click', function () { msg.textContent = 'Taking you to the email screen'; });
})();
</script>
