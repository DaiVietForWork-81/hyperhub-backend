<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var tip = root.querySelector('.u076-tip'), lab = tip.querySelector('span'), kbd = tip.querySelector('kbd'), arr = tip.querySelector('.u076-arr');
  var btns = Array.prototype.slice.call(root.querySelectorAll('[data-t]'));
  var uid = 'u076' + Math.random().toString(36).slice(2, 7);
  var showT = 0, hideT = 0, warm = false, coolT = 0, cur = null;
  tip.id = uid;
  function place(b, glide) {
    lab.textContent = b.getAttribute('data-t'); kbd.textContent = b.getAttribute('data-k') || '';
    var below = !!b.closest('.top');
    tip.classList.toggle('below', below); tip.classList.toggle('above', !below);
    var bx = b.offsetLeft, by = b.offsetTop;
    var w = tip.offsetWidth, h = tip.offsetHeight, cx = bx + b.offsetWidth / 2;
    var x = Math.max(8, Math.min(root.clientWidth - w - 8, cx - w / 2));
    var y = below ? by + b.offsetHeight + 8 : by - h - 8;
    tip.classList.toggle('glide', glide);
    tip.style.setProperty('--x', x + 'px'); tip.style.setProperty('--y', y + 'px');
    arr.style.setProperty('--ax', (cx - x) + 'px');
  }
  function show(b) {
    clearTimeout(hideT); clearTimeout(coolT); clearTimeout(showT);
    var go = function () {
      var was = tip.classList.contains('on');
      place(b, was);
      tip.classList.add('on'); warm = true; cur = b;
      btns.forEach(function (x) { x.removeAttribute('aria-describedby'); });
      b.setAttribute('aria-describedby', uid);
    };
    if (warm) go(); else showT = setTimeout(function () { if (root.isConnected) go(); }, 450);
  }
  function hide(now) {
    clearTimeout(showT);
    var go = function () {
      tip.classList.remove('on'); cur = null;
      coolT = setTimeout(function () { warm = false; }, 300);
    };
    if (now) go(); else hideT = setTimeout(go, 80);
  }
  btns.forEach(function (b) {
    b.addEventListener('pointerenter', function (e) { if (e.pointerType === 'mouse') show(b); });
    b.addEventListener('pointerleave', function (e) { if (e.pointerType === 'mouse') hide(); });
    b.addEventListener('focus', function () { if (b.matches(':focus-visible')) { warm = true; show(b); } });
    b.addEventListener('blur', function () { hide(); });
    b.addEventListener('click', function () {
      if (b.hasAttribute('aria-pressed')) b.setAttribute('aria-pressed', String(b.getAttribute('aria-pressed') !== 'true'));
      hide(true);
    });
  });
  root.addEventListener('keydown', function (e) { if (e.key === 'Escape' && cur) { e.preventDefault(); hide(true); } });
})();
</script>
