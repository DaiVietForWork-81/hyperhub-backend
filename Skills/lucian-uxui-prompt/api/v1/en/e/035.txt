<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var num = root.querySelector('.u035-num');
  var real = root.querySelector('.u035-real');
  var sub = root.querySelector('.u035-sub');
  var cur = root.querySelector('.u035-cur');
  var send = root.querySelector('.u035-send');
  var BAL = 1284000, RATE = 1380;
  var won = 250000, usd = false, prev = '';
  function draw() {
    var v = usd ? (won / RATE) : won;
    var s = usd ? v.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) : v.toLocaleString('en-US');
    if (!won) s = '0';
    var p = prev;
    num.innerHTML = '';
    var off = s.length - p.length;
    Array.from(s).forEach(function (ch, i) {
      var sp = document.createElement('span');
      sp.textContent = ch;
      if (ch === ',' || ch === '.') sp.className = 'c';
      if (!won) sp.classList.add('ph');
      if (ch !== ',' && p[i - off] !== ch) sp.classList.add('new');
      num.appendChild(sp);
    });
    prev = s;
    var over = won > BAL;
    root.classList.toggle('is-over', over);
    sub.textContent = over ? '₩' + (won - BAL).toLocaleString('en-US') + ' over your balance' : 'Balance after: ₩' + (BAL - won).toLocaleString('en-US');
    send.setAttribute('aria-disabled', String(over || !won));
    real.value = String(usd ? Math.round(won / RATE) : won);
  }
  function shake() { root.classList.remove('is-shake'); void root.offsetWidth; root.classList.add('is-shake'); }
  real.addEventListener('focus', function () { root.classList.add('is-focus'); });
  real.addEventListener('blur', function () { root.classList.remove('is-focus'); });
  real.addEventListener('input', function () {
    var digits = real.value.replace(/\D/g, '').replace(/^0+/, '').slice(0, 10);
    var was = won > BAL;
    won = digits ? parseInt(digits, 10) : 0;
    if (usd) won = Math.round(won * RATE);
    draw();
    if (!was && won > BAL) shake();
  });
  Array.prototype.forEach.call(root.querySelectorAll('.u035-chip'), function (b) {
    b.addEventListener('click', function () {
      var was = won > BAL;
      won = b.hasAttribute('data-all') ? BAL : Math.min(9999999999, won + (+b.dataset.add));
      draw();
      if (!was && won > BAL) shake();
    });
  });
  cur.addEventListener('click', function () {
    usd = !usd;
    root.classList.toggle('is-usd', usd);
    cur.setAttribute('aria-label', 'Switch currency: ' + (usd ? 'dollars' : 'won'));
    prev = '';
    draw();
  });
  send.addEventListener('click', function () {
    if (send.getAttribute('aria-disabled') === 'true') { shake(); return; }
    send.textContent = 'Sent ' + (usd ? '$' + (won / RATE).toFixed(2) : '₩' + won.toLocaleString('en-US'));
    setTimeout(function () { if (root.isConnected) send.textContent = 'Send'; }, 1600);
  });
  draw();
})();
</script>
