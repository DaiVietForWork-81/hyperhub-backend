<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];let v=42;const set=x=>{v=Math.max(25,Math.min(65,Math.round(x)));root.style.setProperty("--split",v+"%");q(".sep").setAttribute("aria-valuenow",v)};q(".sep").onkeydown=e=>{if(["ArrowLeft","ArrowRight","Home","End","Escape"].includes(e.key)){e.preventDefault();set(e.key==="Escape"?42:e.key==="Home"?25:e.key==="End"?65:v+(e.key==="ArrowRight"?3:-3))}};q(".sep").onpointerdown=e=>e.currentTarget.setPointerCapture(e.pointerId);q(".sep").onpointermove=e=>{if(e.currentTarget.hasPointerCapture(e.pointerId)){const b=q(".panes").getBoundingClientRect();set((e.clientX-b.left)/b.width*100)}};})();
