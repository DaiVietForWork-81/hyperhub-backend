<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var btn = root.querySelector('.u012-btn');
  var num = root.querySelector('.u012-num');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var gid = 'u012-g-' + Math.random().toString(36).slice(2, 7);
  root.querySelector('linearGradient').id = gid;
  root.querySelector('.u012-in').style.fill = 'url(#' + gid + ')';
  var n = 128, t = 0;
  var roll = root.querySelector('.u012-roll');
  function fitRoll(on) { roll.style.width = roll.children[on ? 1 : 0].offsetWidth + 'px'; }
  fitRoll(false);
  btn.addEventListener('click', function () {
    var on = !root.classList.contains('is-on');
    root.classList.toggle('is-on', on);
    btn.setAttribute('aria-pressed', String(on));
    btn.setAttribute('aria-label', on ? 'Remove from saved' : 'Save this post');
    n += on ? 1 : -1;
    num.textContent = n;
    fitRoll(on);
    root.classList.remove('is-burst');
    clearTimeout(t);
    if (on && !rm) {
      void root.offsetWidth;
      root.classList.add('is-burst');
      t = setTimeout(function () { root.classList.remove('is-burst'); }, 900);
    }
  });
})();
</script>
