<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var re = root.querySelector('.u079-re'), sks = root.querySelectorAll('.u079-sk');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var t = 0;
  function align() {
    sks.forEach(function (s) {
      var x = 0, el = s.parentNode;
      while (el && el !== root) { x += el.offsetLeft; el = el.offsetParent; }
      s.style.setProperty('--sx', -x + 'px');
    });
  }
  function load() {
    clearTimeout(t);
    root.classList.remove('is-done'); root.classList.add('is-loading'); root.setAttribute('aria-busy', 'true');
    re.setAttribute('aria-label', 'Loading');
    t = setTimeout(function () {
      if (!root.isConnected) return;
      root.classList.remove('is-loading'); root.classList.add('is-done'); root.setAttribute('aria-busy', 'false');
      re.setAttribute('aria-label', 'Reload');
    }, rm ? 400 : 1800);
  }
  re.addEventListener('click', load);
  align(); load();
})();
</script>
