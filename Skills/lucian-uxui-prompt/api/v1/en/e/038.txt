<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var menu = root.querySelector('.u038-menu');
  var panels = root.querySelectorAll('.u038-p');
  var share = root.querySelector('.u038-has');
  var back = root.querySelector('.u038-back');
  var cards = Array.prototype.slice.call(root.querySelectorAll('.u038-card'));
  var target = null;
  function local(cx, cy) {
    var r = root.getBoundingClientRect(), k = r.width / root.offsetWidth || 1;
    return { x: (cx - r.left) / k, y: (cy - r.top) / k };
  }
  function items(p) { return Array.prototype.slice.call(panels[p].querySelectorAll('.u038-mi')); }
  function fit(p) { menu.style.setProperty('--mh', (panels[p].offsetHeight + 12) + 'px'); }
  function openAt(card, x, y, kb) {
    closeAll();
    target = card;
    card.classList.add('is-ctx');
    menu.classList.remove('is-sub');
    fit(0);
    var W = root.offsetWidth, H = root.offsetHeight, mw = 200, mh = panels[0].offsetHeight + 12;
    var left = x + mw > W ? Math.max(0, x - mw) : x, top = y + mh > H ? Math.max(0, y - mh) : y;
    menu.style.left = left + 'px'; menu.style.top = top + 'px';
    menu.style.transformOrigin = (left < x ? 'right' : 'left') + ' ' + (top < y ? 'bottom' : 'top');
    menu.classList.add('is-open');
    if (kb) setTimeout(function () { items(0)[0].focus(); }, 30);
  }
  function sub(on, kb) {
    menu.classList.toggle('is-sub', on);
    share.setAttribute('aria-expanded', String(on));
    fit(on ? 1 : 0);
    if (kb) setTimeout(function () { (on ? back : share).focus(); }, 60);
  }
  function closeAll(refocus) {
    menu.classList.remove('is-open', 'is-sub');
    share.setAttribute('aria-expanded', 'false');
    if (target) { target.classList.remove('is-ctx'); if (refocus) target.focus(); }
    target = null;
  }
  cards.forEach(function (card) {
    card.addEventListener('contextmenu', function (e) { e.preventDefault(); var p = local(e.clientX, e.clientY); openAt(card, p.x, p.y); });
    card.querySelector('.u038-more').addEventListener('click', function (e) {
      e.stopPropagation();
      var b = e.currentTarget;
      openAt(card, card.offsetLeft + b.offsetLeft + 28, card.offsetTop + b.offsetTop + 30, true);
    });
    card.addEventListener('keydown', function (e) {
      if (e.target !== card) return;
      if ((e.shiftKey && e.key === 'F10') || e.key === 'ContextMenu' || e.key === 'Enter') {
        e.preventDefault();
        openAt(card, card.offsetLeft + 40, card.offsetTop + 60, true);
      }
    });
    var lp = 0;
    card.addEventListener('pointerdown', function (e) {
      if (e.pointerType !== 'touch') return;
      var cx = e.clientX, cy = e.clientY;
      lp = setTimeout(function () { var p = local(cx, cy); openAt(card, p.x, p.y); }, 480);
    });
    ['pointerup', 'pointercancel', 'pointermove'].forEach(function (t) { card.addEventListener(t, function () { clearTimeout(lp); }); });
    card.querySelector('.u038-undo button').addEventListener('click', function () { card.classList.remove('is-gone'); card.focus(); });
  });
  [0, 1].forEach(function (p) {
    items(p).forEach(function (m, i, arr) {
      m.addEventListener('click', function () {
        if (m === share) { sub(true, true); return; }
        if (m === back) { sub(false, true); return; }
        var t = target;
        if (m.hasAttribute('data-del') && t) { closeAll(false); t.classList.add('is-gone'); t.querySelector('.u038-undo button').focus(); return; }
        closeAll(true);
      });
      m.addEventListener('keydown', function (e) {
        var d = e.key === 'ArrowDown' ? 1 : e.key === 'ArrowUp' ? -1 : 0;
        if (d) { e.preventDefault(); arr[(i + d + arr.length) % arr.length].focus(); }
        if (e.key === 'ArrowRight' && m === share) { e.preventDefault(); sub(true, true); }
        if (e.key === 'ArrowLeft' && p === 1) { e.preventDefault(); sub(false, true); }
      });
    });
  });
  root.addEventListener('keydown', function (e) { if (e.key === 'Escape' && menu.classList.contains('is-open')) { e.preventDefault(); closeAll(true); } });
  document.addEventListener('pointerdown', function (e) { if (root.isConnected && menu.classList.contains('is-open') && !menu.contains(e.target)) closeAll(false); });
})();
</script>
