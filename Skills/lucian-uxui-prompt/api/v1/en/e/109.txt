<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];const days=['Mon','Tue','Wed','Thu','Fri'];q('.week').innerHTML='<span></span>'+days.map(d=>`<span>${d}</span>`).join('')+Array.from({length:5},(_,r)=>`<span>${9+r}:00</span>`+days.map((d,c)=>`<button class="slot" data-i="${r*5+c}" aria-label="${d} ${9+r}:00" aria-pressed="false"></button>`).join('')).join('');let set=new Set(),paintMode=true,drag=false;function paint(){qa('.slot').forEach(b=>b.setAttribute('aria-pressed',set.has(+b.dataset.i)));q('output').textContent=set.size+'h'}function apply(b){if(!b)return;const i=+b.dataset.i;paintMode?set.add(i):set.delete(i);paint()}qa('.slot').forEach(b=>{b.onpointerdown=e=>{drag=true;paintMode=!set.has(+b.dataset.i);b.setPointerCapture(e.pointerId);apply(b)};b.onpointermove=e=>{if(drag&&b.hasPointerCapture(e.pointerId)){const el=document.elementFromPoint(e.clientX,e.clientY);if(el?.classList.contains('slot'))apply(el)}};b.onpointerup=()=>drag=false;b.onpointercancel=()=>drag=false;b.onclick=e=>{if(e.detail===0){paintMode=!set.has(+b.dataset.i);apply(b)}}});root.onkeydown=e=>{if(e.key==='Escape'){set.clear();drag=false;paint()}};})();
</script>
