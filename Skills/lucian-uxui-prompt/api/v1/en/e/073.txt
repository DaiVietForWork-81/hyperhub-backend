<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var msgs = root.querySelector('.u073-msgs'), segs = root.querySelector('.u073-segs'), bar = root.querySelector('.u073-bar');
  var bell = root.querySelector('.u073-bell'), x = root.querySelector('.u073-x'), nx = root.querySelector('.u073-nx');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var D = [['Command palette', 'Find any command with ⌘K'], ['Crisper dark mode', 'Both themes look sharper'], ['October meetup', 'Signups open · first 50']];
  var els = D.map(function (d) { var m = document.createElement('div'); m.className = 'u073-m'; m.innerHTML = '<b></b><span></span>'; m.firstChild.textContent = d[0]; m.lastChild.textContent = d[1]; msgs.appendChild(m); return m; });
  var bars = D.map(function () { var i = document.createElement('i'); segs.appendChild(i); return i; });
  var cur = 0, el = 0, last = performance.now(), paused = false, closed = false, DUR = 3800;
  function show(i) {
    var prev = els[cur];
    cur = (i + D.length) % D.length;
    els.forEach(function (m, k) { m.classList.toggle('on', k === cur); m.classList.toggle('up', m === prev && k !== cur); m.setAttribute('aria-hidden', String(k !== cur)); });
    setTimeout(function () { prev.classList.remove('up'); }, 600);
    bars.forEach(function (b, k) { b.style.setProperty('--f', k < cur ? 1 : 0); });
    el = 0;
  }
  (function loop(now) {
    if (!root.isConnected) return;
    var dt = now - last; last = now;
    if (!paused && !closed && !rm) {
      el += dt;
      bars[cur].style.setProperty('--f', Math.min(1, el / DUR).toFixed(3));
      if (el >= DUR) show(cur + 1);
    }
    requestAnimationFrame(loop);
  })(last);
  bar.addEventListener('pointerenter', function () { paused = true; });
  bar.addEventListener('pointerleave', function () { paused = bar.contains(document.activeElement); });
  bar.addEventListener('focusin', function () { paused = true; });
  bar.addEventListener('focusout', function (e) { if (!bar.contains(e.relatedTarget)) paused = false; });
  nx.addEventListener('click', function () { show(cur + 1); });
  x.addEventListener('click', function () {
    closed = true; root.classList.add('is-closed');
    root.querySelector('.u073-ban').inert = true; bell.tabIndex = 0;
    setTimeout(function () { if (root.isConnected && closed) bell.focus({ preventScroll: true }); }, 300);
  });
  bell.addEventListener('click', function () {
    closed = false; root.classList.remove('is-closed');
    root.querySelector('.u073-ban').inert = false; bell.tabIndex = -1;
    show(0); x.focus({ preventScroll: true });
  });
  show(0);
})();
</script>
