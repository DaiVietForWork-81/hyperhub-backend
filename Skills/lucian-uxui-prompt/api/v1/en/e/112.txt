<div class="ux u112"><div class="steps"></div><div class="transport"><span>120 BPM</span><button class="play" aria-label="Play steps" aria-pressed="false">▶</button></div></div>
<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];let active=new Set([0,4,10,14,16,20]),timer=null,step=-1;const rows=['Kick','Snare','Hat'];q('.steps').innerHTML=rows.map((r,j)=>`<span>${r}</span>`+Array.from({length:8},(_,i)=>`<button class="pad" data-i="${j*8+i}" aria-pressed="false" aria-label="${r} step ${i+1}"></button>`).join('')).join('');function paint(){qa('.pad').forEach(b=>{b.setAttribute('aria-pressed',active.has(+b.dataset.i));b.classList.toggle('playhead',+b.dataset.i%8===step)})}function stop(){clearInterval(timer);timer=null;step=-1;q('.play').textContent='▶';q('.play').setAttribute('aria-label','Play steps');q('.play').setAttribute('aria-pressed',false);paint()}qa('.pad').forEach(b=>b.onclick=()=>{const i=+b.dataset.i;active.has(i)?active.delete(i):active.add(i);paint()});q('.play').onclick=()=>{if(timer){stop();return}q('.play').textContent='Ⅱ';q('.play').setAttribute('aria-label','Stop playback');q('.play').setAttribute('aria-pressed',true);step=0;paint();timer=setInterval(()=>{if(!root.isConnected){stop();return}step=(step+1)%8;paint()},250)};root.onkeydown=e=>{if(e.key==='Escape'){stop();active=new Set([0,4,10,14,16,20]);paint()}};document.addEventListener('visibilitychange',()=>{if(document.hidden)stop()});paint();})();
</script>
