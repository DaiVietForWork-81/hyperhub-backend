<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var form = root.querySelector('.u062-login'), hi = root.querySelector('.u062-hi');
  var pwF = root.querySelector('.u062-pw'), pw = pwF.querySelector('input'), err = pwF.querySelector('.u062-err span');
  var eye = root.querySelector('.u062-eye'), caps = root.querySelector('.u062-caps'), go = root.querySelector('.u062-go');
  var back = hi.querySelector('button');
  var uid = 'u062' + Math.random().toString(36).slice(2, 7);
  root.querySelectorAll('.u062-login .u062-f').forEach(function (f, i) {
    var inp = f.querySelector('input'); inp.id = uid + i; f.querySelector('label').htmlFor = inp.id;
  });
  err.id = uid + 'e'; pw.setAttribute('aria-describedby', err.id);
  var left = 3, busy = false, t = 0;
  eye.addEventListener('click', function () {
    var on = eye.getAttribute('aria-pressed') !== 'true';
    eye.setAttribute('aria-pressed', String(on));
    eye.setAttribute('aria-label', on ? 'Hide password' : 'Show password');
    pw.type = on ? 'text' : 'password';
  });
  pw.addEventListener('keyup', function (e) { if (e.getModifierState) caps.classList.toggle('on', e.getModifierState('CapsLock')); });
  pw.addEventListener('input', function () { pwF.classList.remove('bad'); pw.removeAttribute('aria-invalid'); });
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    if (busy) return;
    busy = true;
    go.classList.add('busy');
    go.setAttribute('aria-label', 'Checking');
    t = setTimeout(function () {
      if (!root.isConnected) return;
      go.classList.remove('busy');
      if (pw.value === 'lumen2026') {
        go.classList.add('ok');
        go.setAttribute('aria-label', 'Signed in');
        t = setTimeout(function () {
          if (!root.isConnected) return;
          root.classList.add('is-in');
          form.inert = true; hi.setAttribute('aria-hidden', 'false'); back.tabIndex = 0;
          back.focus({ preventScroll: true });
        }, 650);
        return;
      }
      busy = false;
      go.removeAttribute('aria-label');
      left = Math.max(0, left - 1);
      err.textContent = left ? 'Wrong password · ' + left + (left === 1 ? ' try left' : ' tries left') : 'Try again later or reset your password';
      pwF.classList.add('bad');
      pw.setAttribute('aria-invalid', 'true');
      root.classList.remove('shake'); void root.offsetWidth; root.classList.add('shake');
      pw.select();
    }, 1100);
  });
  root.addEventListener('animationend', function (e) { if (e.target === root) root.classList.remove('shake'); });
  back.addEventListener('click', function () {
    clearTimeout(t);
    root.classList.remove('is-in');
    form.inert = false; hi.setAttribute('aria-hidden', 'true'); back.tabIndex = -1;
    go.classList.remove('ok'); go.removeAttribute('aria-label');
    busy = false; left = 3; pw.value = ''; pwF.classList.remove('bad');
    pw.focus({ preventScroll: true });
  });
  root.querySelector('.u062-row a').addEventListener('click', function (e) { e.preventDefault(); });
})();
</script>
