<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var dial = root.querySelector('.u067-dial'), time = root.querySelector('.u067-time'), p = root.querySelector('.u067-p');
  var inB = root.querySelector('.u067-in'), mail = root.querySelector('.u067-mail');
  var LOCK = 15, t0 = 0, raf = 0, shown = '';
  function digits(str) {
    if (str === shown) return;
    if (time.children.length !== str.length) {
      time.innerHTML = str.split('').map(function (c) {
        if (c < '0' || c > '9') return '<b>' + c + '</b>';
        var h = ''; for (var n = 0; n < 10; n++) h += '<i>' + n + '</i>';
        return '<span style="--d:' + c + '">' + h + '</span>';
      }).join('');
    } else str.split('').forEach(function (c, k) { if (c >= '0' && c <= '9') time.children[k].style.setProperty('--d', c); });
    shown = str;
  }
  function tick(now) {
    if (!root.isConnected) return;
    var left = Math.max(0, LOCK - (now - t0) / 1000), s = Math.ceil(left);
    dial.style.setProperty('--p', (left / LOCK).toFixed(4));
    digits('0:' + ('0' + s).slice(-2));
    dial.setAttribute('aria-label', s + ' seconds left');
    if (left <= 0) { unlock(); return; }
    raf = requestAnimationFrame(tick);
  }
  function unlock() {
    root.classList.add('is-open');
    inB.setAttribute('aria-disabled', 'false');
    p.innerHTML = 'For your security, take a moment<br>to check your password this time.';
  }
  function lock() {
    cancelAnimationFrame(raf);
    root.classList.remove('is-open');
    inB.setAttribute('aria-disabled', 'true');
    mail.classList.remove('sent'); mail.textContent = 'Email a reset link';
    p.innerHTML = 'Locked after 5 wrong passwords.<br>You can sign in again when time is up.';
    t0 = performance.now();
    raf = requestAnimationFrame(tick);
  }
  inB.addEventListener('click', function () { if (inB.getAttribute('aria-disabled') === 'false') lock(); });
  mail.addEventListener('click', function () {
    mail.classList.add('sent');
    mail.textContent = 'Sent to seoyeon@example.com';
  });
  digits('0:15');
  lock();
})();
</script>
