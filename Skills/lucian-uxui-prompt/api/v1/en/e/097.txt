<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var svg = root.querySelector('svg'), center = root.querySelector('.u097-c'), cs = center.querySelector('small'), cb = center.querySelector('b'), cp = center.querySelector('span'), lg = root.querySelector('.u097-lg');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var D = [['Food', 482000, '#FF8A3D'], ['Shopping', 316000, '#FF6B8B'], ['Transit', 128000, '#3D8BFF'], ['Subscriptions', 89000, '#7B61FF'], ['Other', 65000, '#22C3A6']];
  var R = 80, C = 2 * Math.PI * R, GAP = 3;
  var on = D.map(function () { return true; }), fr = D.map(function () { return 0; }), act = -1, tw = 0, shownTotal = 0, cnt = 0;
  var segs = D.map(function (d, i) {
    var c = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    c.setAttribute('class', 'u097-sg'); c.setAttribute('cx', 108); c.setAttribute('cy', 108); c.setAttribute('r', R);
    c.style.stroke = d[2];
    c.addEventListener('pointerenter', function () { focus(i); });
    c.addEventListener('click', function () { toggle(i); });
    svg.appendChild(c);
    return c;
  });
  var items = D.map(function (d, i) {
    var b = document.createElement('button');
    b.type = 'button'; b.className = 'u097-it'; b.style.setProperty('--c', d[2]); b.setAttribute('aria-pressed', 'true');
    b.innerHTML = '<i aria-hidden="true"></i><span></span><small></small>';
    b.querySelector('span').textContent = d[0];
    b.querySelector('small').textContent = '₩' + Math.round(d[1] / 1000) + 'K';
    b.setAttribute('aria-label', d[0] + ' ₩' + d[1].toLocaleString('ko-KR') + ', included');
    b.addEventListener('pointerenter', function () { focus(i); });
    b.addEventListener('focus', function () { focus(i); });
    b.addEventListener('click', function () { toggle(i); });
    lg.appendChild(b);
    return b;
  });
  svg.addEventListener('pointerleave', function () { focus(-1); });
  lg.addEventListener('pointerleave', function () { if (!lg.contains(document.activeElement)) focus(-1); });
  lg.addEventListener('focusout', function (e) { if (!lg.contains(e.relatedTarget)) focus(-1); });
  function total() { return D.reduce(function (s, d, i) { return s + (on[i] ? d[1] : 0); }, 0); }
  function won(v) { return '₩' + Math.round(v).toLocaleString('ko-KR'); }
  function paint() {
    var acc = 0;
    segs.forEach(function (c, i) {
      var len = fr[i] * C, vis = Math.max(0, len - GAP);
      c.setAttribute('stroke-dasharray', vis.toFixed(2) + ' ' + C.toFixed(2));
      c.setAttribute('stroke-dashoffset', (-acc - GAP / 2).toFixed(2));
      var mid = (acc + len / 2) / C * Math.PI * 2;
      c.style.transform = i === act && on[i] ? 'translate(' + (Math.cos(mid) * 7).toFixed(2) + 'px,' + (Math.sin(mid) * 7).toFixed(2) + 'px)' : '';
      acc += len;
    });
  }
  function morph(stagger) {
    cancelAnimationFrame(tw);
    var t = total(), to = D.map(function (d, i) { return on[i] ? d[1] / t : 0; }), from = fr.slice(), t0 = performance.now(), dur = rm ? 1 : stagger ? 1100 : 600;
    (function f(n) {
      if (!root.isConnected) return;
      var k = Math.min(1, (n - t0) / dur);
      fr = from.map(function (v, i) {
        var kk = stagger ? Math.max(0, Math.min(1, (k - i * 0.1) / 0.6)) : k;
        var e = kk < .5 ? 4 * kk * kk * kk : 1 - Math.pow(-2 * kk + 2, 3) / 2;
        return v + (to[i] - v) * e;
      });
      paint();
      if (k < 1) tw = requestAnimationFrame(f);
    })(t0);
  }
  function rollTotal(to) {
    var from = shownTotal, t0 = performance.now(), dur = rm ? 1 : 600, my = ++cnt;
    (function f(n) {
      if (!root.isConnected || my !== cnt || act >= 0) return;
      var k = Math.min(1, (n - t0) / dur), e = 1 - Math.pow(1 - k, 3);
      shownTotal = from + (to - from) * e; cb.textContent = won(shownTotal);
      if (k < 1) requestAnimationFrame(f);
    })(t0);
  }
  function label() {
    center.classList.remove('sw'); void center.offsetWidth; center.classList.add('sw');
    if (act >= 0 && on[act]) {
      cs.textContent = D[act][0]; cb.textContent = won(D[act][1]);
      cp.textContent = Math.round(D[act][1] / total() * 100) + '% of total';
    } else {
      var n = on.filter(Boolean).length;
      cs.textContent = n === D.length ? 'Total spent' : n + ' selected';
      cb.textContent = won(shownTotal); cp.textContent = n === D.length ? '−8% vs last month' : 'of ' + won(D.reduce(function (s, d) { return s + d[1]; }, 0)) + ' total';
      rollTotal(total());
    }
  }
  function focus(i) {
    if (i === act) return;
    act = i;
    root.classList.toggle('has', i >= 0 && on[i]);
    segs.forEach(function (c, k) { c.classList.toggle('on', k === i && on[k]); });
    items.forEach(function (b, k) { b.classList.toggle('on', k === i); });
    paint(); label();
  }
  function toggle(i) {
    if (on[i] && on.filter(Boolean).length === 1) { items[i].animate([{ transform: 'translateX(0)' }, { transform: 'translateX(-5px)' }, { transform: 'translateX(4px)' }, { transform: 'none' }], { duration: 300 }); return; }
    on[i] = !on[i];
    items[i].setAttribute('aria-pressed', String(on[i]));
    items[i].setAttribute('aria-label', D[i][0] + ' ₩' + D[i][1].toLocaleString('ko-KR') + ', ' + (on[i] ? 'included' : 'excluded'));
    morph(false);
    act = -2; focus(on[i] ? i : -1);
  }
  shownTotal = 0; label();
  morph(!rm);
})();
</script>
