<div class="ux u171"><div class="pool" aria-label="Allocation of 12 cores in total"></div><div class="teams"></div><div class="status" aria-live="polite"></div></div>
<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),names=['Research','Product','Platform'],colors=['#426F90','#6285A2','#8BA2B3'];let vals=[4,3,2];q('.teams').innerHTML=names.map((n,i)=>`<label class="team"><span>${n}</span><input type="number" min="0" max="12" step="1" value="${vals[i]}" data-k="${i}" aria-label="${n} cores"></label>`).join('');q('.pool').innerHTML=vals.map((v,i)=>`<i style="--n:${v};--c:${colors[i]}"></i>`).join('');function draw(){let total=vals.reduce((a,b)=>a+b,0);root.querySelectorAll('.pool i').forEach((el,i)=>{el.style.setProperty('--n',vals[i])});root.querySelectorAll('input').forEach((el,i)=>{el.value=vals[i];el.max=12-total+vals[i]});q('.status').textContent=`${total}/12 cores · ${12-total} cores left`;root.dataset.state=vals.join(',')}q('.teams').oninput=e=>{let el=e.target;if(el.tagName!=='INPUT')return;let i=+el.dataset.k,remaining=12-vals.reduce((sum,v,j)=>sum+(j===i?0:v),0);vals[i]=Math.max(0,Math.min(remaining,Math.round(+el.value||0)));draw()};draw()})();
</script>
