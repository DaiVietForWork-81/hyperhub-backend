<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var btn = root.querySelector('.u005-btn');
  var live = root.querySelector('.u005-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var timers = [];
  function later(fn, ms) { timers.push(setTimeout(function () { if (root.isConnected) fn(); }, ms)); }
  btn.addEventListener('click', function () {
    if (root.dataset.state !== 'idle') return;
    root.dataset.state = 'fly';
    live.textContent = 'Message sent';
    later(function () { root.dataset.state = 'sent'; }, rm ? 50 : 1000);
    later(function () { root.dataset.state = 'back'; }, rm ? 900 : 2300);
    later(function () { root.dataset.state = 'idle'; live.textContent = ''; }, rm ? 1000 : 3000);
  });
})();
</script>
