<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var btn = root.querySelector('.u003-btn');
  var file = root.querySelector('.u003-file');
  var done = root.querySelector('.u003-done');
  var undo = root.querySelector('.u003-undo');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var HOLD = 1200;
  var p = 0, raf = 0, last = 0, holding = false, finished = false;

  function set(v) { p = v; btn.style.setProperty('--p', v.toFixed(4)); }
  function tick(now) {
    if (!root.isConnected) return;
    var dt = now - (last || now);
    last = now;
    if (holding) set(Math.min(1, p + dt / HOLD));
    else set(Math.max(0, p - dt / (HOLD / 2.4)));
    btn.style.setProperty('--shake', holding && p > 0.55 && !rm ? 'running' : 'paused');
    if (p >= 1 && holding) { complete(); return; }
    if (holding || p > 0) raf = requestAnimationFrame(tick);
    else { raf = 0; last = 0; }
  }
  function start() {
    if (finished) return;
    holding = true;
    btn.classList.add('is-holding');
    if (!raf) { last = 0; raf = requestAnimationFrame(tick); }
  }
  function stop() {
    holding = false;
    btn.classList.remove('is-holding');
    if (!raf && p > 0) raf = requestAnimationFrame(tick);
  }
  function complete() {
    holding = false; finished = true; raf = 0; last = 0;
    btn.classList.remove('is-holding');
    file.classList.add('is-gone');
    btn.classList.add('is-gone');
    btn.setAttribute('tabindex', '-1');
    done.classList.add('is-on');
    undo.focus({ preventScroll: true });
  }
  function restore() {
    finished = false;
    set(0);
    file.classList.remove('is-gone');
    btn.classList.remove('is-gone');
    btn.removeAttribute('tabindex');
    done.classList.remove('is-on');
    btn.focus({ preventScroll: true });
  }
  btn.addEventListener('pointerdown', function (e) { if (e.button === 0) { btn.setPointerCapture(e.pointerId); start(); } });
  btn.addEventListener('pointerup', stop);
  btn.addEventListener('pointercancel', stop);
  btn.addEventListener('lostpointercapture', stop);
  btn.addEventListener('contextmenu', function (e) { e.preventDefault(); });
  btn.addEventListener('keydown', function (e) {
    if (e.key === ' ' || e.key === 'Enter') { e.preventDefault(); if (!e.repeat) start(); }
  });
  btn.addEventListener('keyup', function (e) { if (e.key === ' ' || e.key === 'Enter') stop(); });
  btn.addEventListener('blur', stop);
  btn.addEventListener('click', function (e) { e.preventDefault(); });
  undo.addEventListener('click', restore);
})();
</script>
