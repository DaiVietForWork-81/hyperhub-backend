(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),bids=[[99,12],[98,25],[97,40]],asks=[[101,8],[102,16],[103,31]],xs=[8,54,100,150,200,245,292];function draw(){let n=+q('.range').value,price=97+n,side=n<3?'Bid':'Ask',k=n<3?2-n:n-4,d=n<3?bids:asks,x=xs[n];q('.cursor').setAttribute('x1',0);q('.cursor').setAttribute('x2',0);q('.cursor').style.transform=`translateX(${x}px)`;q('.status').textContent=n===3?'Mid price 100.00':`${side} ${price}.00 · cumulative ${d.slice(0,k+1).reduce((sum,row)=>sum+row[1],0)}`;q('.range').setAttribute('aria-valuetext',q('.status').textContent);root.dataset.state=String(n)}q('.range').oninput=draw;q('.depth').onpointerdown=e=>{let r=q('.depth').getBoundingClientRect(),x=(e.clientX-r.left)/r.width*300;q('.range').value=xs.reduce((best,v,i)=>Math.abs(v-x)<Math.abs(xs[best]-x)?i:best,0);draw()};draw()})();
