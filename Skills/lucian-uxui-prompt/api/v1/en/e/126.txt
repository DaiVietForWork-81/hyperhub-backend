<script>
(()=>{const root=document.currentScript.previousElementSibling,$=s=>root.querySelector(s),$$=s=>[...root.querySelectorAll(s)],esc=v=>String(v).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));const motion=(node,frames,opts={})=>{if(node?.animate&&!matchMedia('(prefers-reduced-motion:reduce)').matches)node.animate(frames,{duration:220,easing:'cubic-bezier(.2,.8,.2,1)',...opts})};const source=$('textarea');function render(){let h='',list=false;for(const l of source.value.split('\n')){const t=esc(l).replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>');if(l.startsWith('- ')){if(!list){h+='<ul>';list=true}h+='<li>'+t.slice(2)+'</li>'}else{if(list){h+='</ul>';list=false}h+=l.startsWith('# ')?'<h2>'+t.slice(2)+'</h2>':l?'<p>'+t+'</p>':''}}$('article').innerHTML=h+(list?'</ul>':'')}function edit(on){source.hidden=!on;$('article').hidden=on;$('[data-edit]').textContent=on?'Preview':'Edit';$('[data-edit]').setAttribute('aria-pressed',String(on));if(on)source.focus();else motion($('article'),[{transform:'translateY(4px)',opacity:.8},{transform:'none',opacity:1}])}source.oninput=render;source.onkeydown=e=>{if(e.key==='Escape')edit(false)};$('[data-edit]').onclick=()=>edit(source.hidden);render();})();
</script>
