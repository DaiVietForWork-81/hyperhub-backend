<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];let v=50;function set(n){v=Math.max(5,Math.min(95,Math.round(n)));root.style.setProperty('--pos',v+'%');q('.gradient').style.background=`linear-gradient(90deg,#536f9d 0%,#ad899e ${v}%,#d5ad82 100%)`;q('.stop').setAttribute('aria-valuenow',v);q('output').textContent=v+'%'}q('.stop').onkeydown=e=>{if(['ArrowLeft','ArrowRight','Home','End','Escape'].includes(e.key)){e.preventDefault();set(e.key==='Escape'?50:e.key==='Home'?5:e.key==='End'?95:v+(e.key==='ArrowRight'?5:-5))}};q('.stop').onpointerdown=e=>e.currentTarget.setPointerCapture(e.pointerId);q('.stop').onpointermove=e=>{if(e.currentTarget.hasPointerCapture(e.pointerId)){const r=q('.track').getBoundingClientRect();set((e.clientX-r.left)/r.width*100)}};})();
