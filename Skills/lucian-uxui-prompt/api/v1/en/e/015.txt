<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var opts = Array.prototype.slice.call(root.querySelectorAll('.u015-opt'));
  var frame = root.querySelector('.u015-frame');
  var sel = root.querySelector('.u015-sel');
  var sum = root.querySelector('.u015-sum');
  function place(o) {
    frame.style.top = o.offsetTop + 'px';
    frame.style.height = o.offsetHeight + 'px';
  }
  function choose(o, focus) {
    opts.forEach(function (x) {
      var on = x === o;
      x.setAttribute('aria-checked', String(on));
      x.tabIndex = on ? 0 : -1;
    });
    place(o);
    var name = o.querySelector('.u015-txt b').firstChild.textContent.trim();
    var p = +o.dataset.p;
    sel.textContent = name + ' plan';
    sum.textContent = p ? '$' + (p / 1000) + ' / month' : 'Free';
    if (focus) o.focus();
  }
  opts.forEach(function (o, i) {
    o.addEventListener('click', function () { choose(o); });
    o.addEventListener('keydown', function (e) {
      var d = (e.key === 'ArrowDown' || e.key === 'ArrowRight') ? 1 : (e.key === 'ArrowUp' || e.key === 'ArrowLeft') ? -1 : 0;
      if (d) { e.preventDefault(); choose(opts[(i + d + opts.length) % opts.length], true); }
    });
  });
  frame.style.transition = 'none';
  choose(opts[1]);
  requestAnimationFrame(function () { requestAnimationFrame(function () { frame.style.transition = ''; }); });
})();
</script>
