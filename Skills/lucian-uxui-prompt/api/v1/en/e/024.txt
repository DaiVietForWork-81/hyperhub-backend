  box-shadow:0 0 0 2.5px #14B8A6,0 6px 14px -4px rgba(0,60,50,.4);transition:transform .35s var(--ux-spring),box-shadow .3s var(--ux-ease)}

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var track = root.querySelector('.u024-track');
  var hist = root.querySelector('.u024-hist');
  var val = root.querySelector('.u024-val');
  var roll = root.querySelector('.u024-roll');
  var handles = Array.prototype.slice.call(root.querySelectorAll('.u024-h'));
  var STEP = 2;
  var v = { lo: 16, hi: 60 };
  var COUNTS = [2, 3, 4, 6, 7, 9, 11, 14, 16, 15, 13, 12, 10, 11, 9, 8, 6, 5, 6, 4, 3, 3, 2, 1];
  var MAXC = 16;
  var bars = COUNTS.map(function (c) {
    var b = document.createElement('span');
    b.className = 'u024-bar';
    b.style.setProperty('--h', Math.max(8, c / MAXC * 100).toFixed(1));
    hist.appendChild(b);
    return b;
  });
  var won = function (p) { return Math.round((30000 + p / 100 * 150000) / 1000) * 1000; };
  var fmt = function (n) { return '$' + Math.round(n / 1000); };
  var ci = 0, shown = '';
  function setCount(n) {
    if (String(n) === shown) return;
    var up = !shown || n > +shown;
    shown = String(n);
    var cur = roll.children[ci], nxt = roll.children[1 - ci];
    nxt.style.transition = 'none';
    nxt.textContent = n;
    nxt.style.transform = 'translateY(' + (up ? 100 : -100) + '%)'; nxt.style.opacity = '0';
    void nxt.offsetWidth;
    nxt.style.transition = '';
    nxt.style.transform = 'none'; nxt.style.opacity = '1';
    cur.style.transform = 'translateY(' + (up ? -100 : 100) + '%)'; cur.style.opacity = '0';
    ci = 1 - ci;
  }
  function paint() {
    handles[0].style.setProperty('--pos', v.lo);
    handles[1].style.setProperty('--pos', v.hi);
    root.style.setProperty('--a', v.lo);
    root.style.setProperty('--b', v.hi);
    [['lo', 0], ['hi', 1]].forEach(function (x) {
      handles[x[1]].setAttribute('aria-valuenow', String(v[x[0]]));
      handles[x[1]].setAttribute('aria-valuetext', '$' + Math.round(won(v[x[0]]) / 1000));
    });
    val.textContent = fmt(won(v.lo)) + ' – ' + fmt(won(v.hi));
    var n = 0;
    bars.forEach(function (b, i) {
      var c = (i + 0.5) / bars.length * 100;
      var inside = c >= v.lo && c <= v.hi;
      b.classList.toggle('in', inside);
      if (inside) n += COUNTS[i];
    });
    setCount(n);
  }
  function posFromX(x) {
    var r = track.getBoundingClientRect();
    return Math.max(0, Math.min(100, Math.round((x - r.left) / r.width * 100 / STEP) * STEP));
  }
  function drag(h, e) {
    var p = posFromX(e.clientX);
    if (h.dataset.k === 'lo') v.lo = Math.min(p, v.hi - STEP * 3); else v.hi = Math.max(p, v.lo + STEP * 3);
    paint();
  }
  handles.forEach(function (h) {
    h.addEventListener('pointerdown', function (e) { h.setPointerCapture(e.pointerId); h.classList.add('drag'); drag(h, e); });
    h.addEventListener('pointermove', function (e) { if (h.hasPointerCapture(e.pointerId)) drag(h, e); });
    h.addEventListener('pointerup', function () { h.classList.remove('drag'); });
    h.addEventListener('pointercancel', function () { h.classList.remove('drag'); });
    h.addEventListener('keydown', function (e) {
      var d = (e.key === 'ArrowRight' || e.key === 'ArrowUp') ? STEP : (e.key === 'ArrowLeft' || e.key === 'ArrowDown') ? -STEP : 0;
      if (!d) return;
      e.preventDefault();
      if (h.dataset.k === 'lo') v.lo = Math.max(0, Math.min(v.lo + d, v.hi - STEP * 3));
      else v.hi = Math.min(100, Math.max(v.hi + d, v.lo + STEP * 3));
      paint();
    });
  });
  paint();
})();
</script>
