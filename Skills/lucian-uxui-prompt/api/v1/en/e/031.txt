<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var btn = root.querySelector('.u031-btn');
  var list = root.querySelector('.u031-list');
  var hl = root.querySelector('.u031-hl');
  var save = root.querySelector('.u031-save');
  var OPTS = [['KO', '한국어', 'South Korea', 'linear-gradient(140deg,#5B8CFF,#E5484D)'], ['EN', 'English', 'United States', 'linear-gradient(140deg,#3D6BFF,#FF6B6B)'], ['JA', '日本語', 'Japan', 'linear-gradient(140deg,#FF7A93,#F0264D)'], ['ES', 'Español', 'Spain', 'linear-gradient(140deg,#FFB547,#E5484D)'], ['FR', 'Français', 'France', 'linear-gradient(140deg,#5B8CFF,#8B6CFF)']];
  var uid = 'u031-' + Math.random().toString(36).slice(2, 7);
  var sel = 1, act = 1, typed = '', tt = 0;
  list.id = uid;
  btn.setAttribute('aria-controls', uid);
  var items = OPTS.map(function (o, i) {
    var li = document.createElement('li');
    li.className = 'u031-o';
    li.id = uid + '-' + i;
    li.setAttribute('role', 'option');
    li.setAttribute('aria-selected', String(i === sel));
    li.innerHTML = '<span class="u031-flag" style="--c:' + o[3] + '" aria-hidden="true">' + o[0] + '</span>' + o[1] + '<small>' + o[2] + '</small><svg class="u031-ck" viewBox="0 0 18 18" aria-hidden="true"><path d="M4 9.5l3.5 3.5L14 6"/></svg>';
    li.addEventListener('pointermove', function () { if (act !== i) { act = i; mark(); } });
    li.addEventListener('click', function () { choose(i); });
    list.appendChild(li);
    return li;
  });
  function mark() {
    hl.style.setProperty('--i', act);
    list.setAttribute('aria-activedescendant', items[act].id);
  }
  function open() {
    root.classList.add('is-open');
    btn.setAttribute('aria-expanded', 'true');
    list.classList.add('is-open');
    act = sel; mark();
    list.focus({ preventScroll: true });
  }
  function close(refocus) {
    root.classList.remove('is-open');
    btn.setAttribute('aria-expanded', 'false');
    list.classList.remove('is-open');
    if (refocus) btn.focus({ preventScroll: true });
  }
  function choose(i) {
    sel = i;
    items.forEach(function (li, k) { li.setAttribute('aria-selected', String(k === i)); });
    var o = OPTS[i];
    btn.querySelector('.u031-flag').textContent = o[0];
    btn.querySelector('.u031-flag').style.setProperty('--c', o[3]);
    btn.querySelector('.u031-v b').textContent = o[1];
    btn.querySelector('.u031-v small').textContent = o[2];
    var v = btn.querySelector('.u031-v');
    if (v.animate) v.animate([{ transform: 'translateY(8px)', opacity: 0 }, { transform: 'none', opacity: 1 }], { duration: 380, easing: 'cubic-bezier(.34,1.45,.64,1)' });
    close(true);
  }
  btn.addEventListener('click', function () { list.classList.contains('is-open') ? close(true) : open(); });
  btn.addEventListener('keydown', function (e) { if (e.key === 'ArrowDown' || e.key === 'ArrowUp') { e.preventDefault(); open(); } });
  list.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowDown') { e.preventDefault(); act = Math.min(items.length - 1, act + 1); mark(); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); act = Math.max(0, act - 1); mark(); }
    else if (e.key === 'Home') { e.preventDefault(); act = 0; mark(); }
    else if (e.key === 'End') { e.preventDefault(); act = items.length - 1; mark(); }
    else if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); choose(act); }
    else if (e.key === 'Escape') { e.preventDefault(); close(true); }
    else if (e.key === 'Tab') close(false);
    else if (e.key.length === 1) {
      typed += e.key.toLowerCase();
      clearTimeout(tt); tt = setTimeout(function () { typed = ''; }, 600);
      var hit = OPTS.findIndex(function (o) { return o[1].toLowerCase().indexOf(typed) === 0 || o[0].toLowerCase().indexOf(typed) === 0; });
      if (hit >= 0) { act = hit; mark(); }
    }
  });
  list.addEventListener('blur', function () { setTimeout(function () { if (!root.contains(document.activeElement)) close(false); }, 0); });
  document.addEventListener('pointerdown', function (e) { if (root.isConnected && list.classList.contains('is-open') && !root.querySelector('.u031-f').contains(e.target)) close(false); });
  save.addEventListener('click', function () { save.textContent = 'Saved'; setTimeout(function () { if (root.isConnected) save.textContent = 'Save'; }, 1400); });
})();
</script>
