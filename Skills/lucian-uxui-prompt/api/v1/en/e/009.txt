<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var btn = root.querySelector('.u009-btn');
  var sr = root.querySelector('.u009-sr');
  Array.prototype.forEach.call(root.querySelectorAll('.u009-face'), function (face) {
    Array.from(face.dataset.t).forEach(function (ch, i) {
      var s = document.createElement('span');
      s.textContent = ch === ' ' ? ' ' : ch;
      s.dataset.c = ch === ' ' ? ' ' : ch;
      s.style.setProperty('--i', i);
      face.appendChild(s);
    });
  });
  btn.addEventListener('click', function () {
    var on = !root.classList.contains('is-on');
    root.classList.toggle('is-on', on);
    btn.setAttribute('aria-pressed', String(on));
    sr.textContent = on ? 'Browsing' : 'See our work';
  });
})();
</script>
