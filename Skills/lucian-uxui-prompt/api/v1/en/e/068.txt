<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var txt = root.querySelector('.u068-txt'), doc = root.querySelector('.u068-doc'), readMsg = root.querySelector('.u068-read span');
  var all = root.querySelector('.u068-all'), subs = Array.prototype.slice.call(root.querySelectorAll('.u068-list input:not(.u068-all)'));
  var go = root.querySelector('.u068-go');
  var read = false, ts = [];
  txt.addEventListener('scroll', function () {
    var max = txt.scrollHeight - txt.clientHeight, p = max > 0 ? txt.scrollTop / max : 1;
    doc.style.setProperty('--p', Math.min(1, p).toFixed(3));
    if (!read && p > 0.97) {
      read = true;
      root.classList.add('is-read');
      readMsg.textContent = 'All read · You can agree now';
      subs.forEach(function (s) { s.disabled = false; });
      all.disabled = false;
    }
  }, { passive: true });
  function sync() {
    var on = subs.filter(function (s) { return s.checked; }).length;
    all.checked = on === subs.length;
    all.indeterminate = on > 0 && on < subs.length;
    var ok = subs.every(function (s) { return !s.getAttribute('data-req') || s.checked; });
    go.setAttribute('aria-disabled', String(!ok));
  }
  all.addEventListener('change', function () {
    var v = all.checked;
    if (!read) { all.checked = false; txt.animate([{ transform: 'translateX(-5px)' }, { transform: 'translateX(4px)' }, { transform: 'none' }], { duration: 300 }); readMsg.textContent = 'Scroll the terms to the end first'; return; }
    ts.forEach(clearTimeout); ts = [];
    subs.forEach(function (s, i) { ts.push(setTimeout(function () { if (!root.isConnected) return; s.checked = v; sync(); }, i * 90)); });
  });
  subs.forEach(function (s) { s.addEventListener('change', sync); });
  go.addEventListener('click', function () {
    if (go.getAttribute('aria-disabled') === 'true') {
      if (!read) txt.focus();
      go.animate([{ transform: 'translateX(-5px)' }, { transform: 'translateX(4px)' }, { transform: 'none' }], { duration: 300 });
      return;
    }
    go.textContent = 'Welcome!';
  });
})();
</script>
