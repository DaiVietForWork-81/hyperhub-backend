<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];const taken=['A3','B4','B5','C2','D6'];let chosen=new Set();q('.seats').innerHTML=['A','B','C','D'].map(r=>`<span class="row">${r}</span>`+Array.from({length:6},(_,i)=>{const s=r+(i+1);return `<button class="seat" data-seat="${s}" aria-label="${s} ${taken.includes(s)?'booked':'available'}" aria-pressed="false" ${taken.includes(s)?'disabled':''}>${taken.includes(s)?'×':i+1}</button>`}).join('')).join('');function paint(){qa('.seat').forEach(b=>b.setAttribute('aria-pressed',chosen.has(b.dataset.seat)));q('output').textContent=chosen.size?[...chosen].sort().join(' · ')+' / '+'$'+(chosen.size*12):'—'}qa('.seat:not(:disabled)').forEach(b=>b.onclick=()=>{const s=b.dataset.seat;chosen.has(s)?chosen.delete(s):chosen.add(s);paint()});root.onkeydown=e=>{if(e.key==='Escape'){chosen.clear();paint()}};})();
</script>
