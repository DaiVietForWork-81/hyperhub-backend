<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var chip = root.querySelector('.u037-chip');
  var items = Array.prototype.slice.call(root.querySelectorAll('.u037-it'));
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  function setOpen(v, focusFirst) {
    root.classList.toggle('is-open', v);
    chip.setAttribute('aria-expanded', String(v));
    items.forEach(function (it) { it.tabIndex = v ? 0 : -1; });
    if (v && focusFirst) setTimeout(function () { items[0].focus(); }, 60);
  }
  // 커서 아래 항목으로 강조 막대가 미끄러져 갑니다. 위치는 레이아웃 값으로 계산해 배율과 무관합니다
  function hl(it) {
    var list = it.closest('.u037-list');
    Array.prototype.forEach.call(root.querySelectorAll('.u037-list'), function (l) { if (l !== list) l.classList.remove('is-hl'); });
    var bar = list.querySelector('.u037-hl');
    bar.style.setProperty('--y', it.parentNode.offsetTop + 'px');
    bar.style.setProperty('--h', it.offsetHeight + 'px');
    list.classList.add('is-hl');
  }
  items.forEach(function (it, i) {
    it.addEventListener('pointerenter', function () { hl(it); });
    it.addEventListener('focus', function () { hl(it); });
    it.addEventListener('keydown', function (e) {
      var d = e.key === 'ArrowDown' ? 1 : e.key === 'ArrowUp' ? -1 : 0;
      if (d) { e.preventDefault(); items[(i + d + items.length) % items.length].focus(); }
    });
    it.addEventListener('click', function () {
      if (it.dataset.ws) {
        items.forEach(function (x) { if (x.dataset.ws) x.setAttribute('aria-checked', String(x === it)); });
        var ws = chip.querySelector('.u037-ws');
        ws.textContent = it.dataset.ws;
        ws.style.setProperty('--c', it.dataset.c);
        chip.querySelector('b').textContent = it.dataset.name;
        if (ws.animate && !rm) ws.animate([{ transform: 'scale(.6) rotate(-20deg)' }, { transform: 'none' }], { duration: 450, easing: 'cubic-bezier(.34,1.45,.64,1)' });
      }
      setOpen(false);
      chip.focus({ preventScroll: true });
    });
  });
  root.querySelector('.u037-menu').addEventListener('pointerleave', function () {
    Array.prototype.forEach.call(root.querySelectorAll('.u037-list'), function (l) { l.classList.remove('is-hl'); });
  });
  chip.addEventListener('click', function () { setOpen(!root.classList.contains('is-open'), false); });
  chip.addEventListener('keydown', function (e) { if (e.key === 'ArrowUp') { e.preventDefault(); setOpen(true, true); } });
  root.addEventListener('keydown', function (e) { if (e.key === 'Escape' && root.classList.contains('is-open')) { e.preventDefault(); setOpen(false); chip.focus(); } });
  setOpen(true, false);
})();
</script>
