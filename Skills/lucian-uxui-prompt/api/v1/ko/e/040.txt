<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var tog = root.querySelector('.u040-tog');
  var its = Array.prototype.slice.call(root.querySelectorAll('.u040-it'));
  var hl = root.querySelector('.u040-hl');
  var h3 = root.querySelector('.u040-main h3');
  function place(it) { hl.style.setProperty('--y', it.parentNode.offsetTop + 'px'); }
  tog.addEventListener('click', function () {
    var w = !root.classList.contains('is-wide');
    root.classList.toggle('is-wide', w);
    tog.setAttribute('aria-expanded', String(w));
    tog.setAttribute('aria-label', w ? '메뉴 접기' : '메뉴 펼치기');
  });
  its.forEach(function (it, i) {
    it.addEventListener('click', function () {
      its.forEach(function (x) { x.removeAttribute('aria-current'); });
      it.setAttribute('aria-current', 'page');
      place(it);
      // 오른쪽 화면 제목이 살짝 내려앉으며 바뀝니다
      h3.classList.add('swap');
      setTimeout(function () { if (!root.isConnected) return; h3.textContent = it.querySelector('span').textContent; h3.classList.remove('swap'); }, 180);
    });
    it.addEventListener('keydown', function (e) {
      var d = e.key === 'ArrowDown' ? 1 : e.key === 'ArrowUp' ? -1 : 0;
      if (d) { e.preventDefault(); its[(i + d + its.length) % its.length].focus(); }
    });
  });
  place(its[0]);
})();
</script>
