<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var go = root.querySelector('.u081-go'), ring = root.querySelector('.u081-ring'), steps = root.querySelectorAll('.u081-steps span');
  var rec = root.querySelector('.u081-rec'), back = rec.querySelector('button'), sr = root.querySelector('.u081-sr');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var raf = 0, ts = [];
  function later(fn, ms) { ts.push(setTimeout(function () { if (root.isConnected) fn(); }, ms)); }
  // 링은 처음엔 빠르게, 끝에 가까워질수록 천천히 차오릅니다. 단계 표시는 링 진행에 맞춰 켜집니다
  function run() {
    root.classList.add('is-run'); sr.textContent = '결제를 처리하는 중이에요';
    var t0 = performance.now(), DUR = rm ? 600 : 2600, delay = rm ? 0 : 350;
    (function step(now) {
      if (!root.isConnected) return;
      var k = Math.max(0, Math.min(1, (now - t0 - delay) / DUR)), e = 1 - Math.pow(1 - k, 2.2);
      ring.style.setProperty('--p', e.toFixed(4));
      steps.forEach(function (s, i) { s.classList.toggle('on', e >= [0.02, 0.45, 0.999][i]); });
      if (k < 1) { raf = requestAnimationFrame(step); return; }
      root.classList.remove('is-run'); root.classList.add('is-ok'); sr.textContent = '결제를 마쳤어요';
      later(function () { root.classList.add('is-rec'); rec.setAttribute('aria-hidden', 'false'); back.tabIndex = 0; back.focus({ preventScroll: true }); }, rm ? 0 : 900);
    })(t0);
  }
  go.addEventListener('click', run);
  back.addEventListener('click', function () {
    cancelAnimationFrame(raf); ts.forEach(clearTimeout); ts = [];
    root.classList.remove('is-run', 'is-ok', 'is-rec'); rec.setAttribute('aria-hidden', 'true'); back.tabIndex = -1;
    ring.style.setProperty('--p', 0); steps.forEach(function (s) { s.classList.remove('on'); });
    go.focus({ preventScroll: true });
  });
})();
</script>
