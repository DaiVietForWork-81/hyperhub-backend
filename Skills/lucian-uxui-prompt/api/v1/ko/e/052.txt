<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var card = root.querySelector('.u052-card'), flipB = root.querySelector('.u052-flip'), lk = root.querySelector('.u052-lk');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var ang = 0, sx = 0, start = 0, moved = false, down = false, lastX = 0, vx = 0;
  function apply(a) {
    root.style.setProperty('--ry', a + 'deg');
    // 홀로그램 띠는 카드가 도는 만큼 옆으로 흐릅니다
    root.style.setProperty('--sh', (50 + Math.sin(a * Math.PI / 180) * 60).toFixed(1) + '%');
  }
  function settle(a) {
    ang = Math.round(a / 180) * 180;
    apply(ang);
    var back = Math.abs(ang / 180) % 2 === 1;
    card.setAttribute('aria-label', '멤버십 카드, ' + (back ? '뒷면' : '앞면') + '. 눌러서 뒤집기');
    flipB.lastChild.textContent = back ? '앞면 보기' : '뒷면 보기';
  }
  function flip() { settle(ang + 180); }
  // 끌면 손을 따라 돌고, 놓으면 속도와 위치를 보고 가까운 면으로 붙습니다
  card.addEventListener('pointerdown', function (e) {
    if (e.button !== 0) return;
    down = true; moved = false; sx = lastX = e.clientX; start = ang; vx = 0;
    card.setPointerCapture(e.pointerId);
  });
  card.addEventListener('pointermove', function (e) {
    if (!down) return;
    var dx = e.clientX - sx;
    if (!moved && Math.abs(dx) > 6) { moved = true; card.classList.add('drag'); }
    if (!moved) return;
    vx = e.clientX - lastX; lastX = e.clientX;
    apply(start + dx * 0.9);
  });
  function up(e) {
    if (!down) return;
    down = false;
    card.classList.remove('drag');
    if (!moved) return;
    var cur = start + (e.clientX - sx) * 0.9 + vx * 8;
    settle(cur);
  }
  card.addEventListener('pointerup', up);
  card.addEventListener('pointercancel', up);
  card.addEventListener('click', function (e) { if (moved) { e.preventDefault(); moved = false; return; } flip(); });
  // 마우스를 올리면 살짝 앞으로 기울어 입체감을 보여 줍니다
  card.addEventListener('pointerenter', function (e) { if (e.pointerType === 'mouse' && !rm) root.style.setProperty('--tilt', '8deg'); });
  card.addEventListener('pointerleave', function () { root.style.setProperty('--tilt', '0deg'); });
  flipB.addEventListener('click', flip);
  lk.addEventListener('click', function () {
    var on = lk.getAttribute('aria-pressed') !== 'true';
    lk.setAttribute('aria-pressed', String(on));
    lk.lastChild.textContent = on ? '잠금 해제' : '카드 잠금';
    root.classList.toggle('is-locked', on);
  });
  settle(0);
})();
</script>
