<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];const zones=[['서울','Asia/Seoul'],['런던','Europe/London'],['뉴욕','America/New_York']];q('.zones').innerHTML=zones.map(([name])=>`<div class="zone"><span class="city">${name}</span><div class="hours"><span class="work"></span><span class="needle"></span></div><span class="clock"></span></div>`).join('');function paint(){const h=+q('.time').value,date=new Date('2026-10-12T'+String(h).padStart(2,'0')+':00:00Z');let ok=0;qa('.zone').forEach((r,i)=>{const hour=+new Intl.DateTimeFormat('en-GB',{timeZone:zones[i][1],hour:'2-digit',hourCycle:'h23'}).format(date),good=hour>=9&&hour<18;r.querySelector('.clock').innerHTML=String(hour).padStart(2,'0')+':00<small>'+(good?'근무':'휴식')+'</small>';r.style.setProperty('--h',hour/24*100+'%');if(good)ok++});q('.utc').textContent=String(h).padStart(2,'0')+':00';q('.result').textContent='업무 시간에 있는 지역 '+ok+' / 3'}q('.time').oninput=paint;paint();})();
</script>
