<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];let x=.25,y=.1,anim;function paint(){q('.curve').setAttribute('d',`M10 190C${10+x*180} ${190-y*180} 55 10 190 10`);q('.guide').setAttribute('d',`M10 190L${10+x*180} ${190-y*180}`);root.style.setProperty('--x',(5+x*90)+'%');root.style.setProperty('--y',(95-y*90)+'%');q('output').textContent=`cubic-bezier(${x.toFixed(2)}, ${y.toFixed(2)}, 0.25, 1)`}function run(){anim?.cancel();anim=q('.ball').animate([{transform:'translateX(0)'},{transform:`translateX(${q('.runway').clientWidth-26}px)`}],{duration:matchMedia('(prefers-reduced-motion: reduce)').matches?1:1000,easing:`cubic-bezier(${x},${y},.25,1)`,fill:'forwards'})}function set(a,b){x=Math.max(0,Math.min(1,a));y=Math.max(0,Math.min(1,b));paint()}q('.handle').onkeydown=e=>{let d={ArrowLeft:[-.05,0],ArrowRight:[.05,0],ArrowUp:[0,.05],ArrowDown:[0,-.05]}[e.key];if(d){e.preventDefault();set(x+d[0],y+d[1]);run()}if(e.key==='Escape'){anim?.cancel();set(.25,.1)}};q('.handle').onpointerdown=e=>e.currentTarget.setPointerCapture(e.pointerId);q('.handle').onpointermove=e=>{if(e.currentTarget.hasPointerCapture(e.pointerId)){let r=q('.graph').getBoundingClientRect();set(((e.clientX-r.left)/r.width-.05)/.9,(.95-(e.clientY-r.top)/r.height)/.9)}};q('.handle').onpointerup=run;paint();})();
</script>
