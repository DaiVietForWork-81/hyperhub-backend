<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];const tree={name:'작업실',children:[{name:'프로젝트',children:[{name:'오로라',children:[{name:'원고',children:[]},{name:'표지',children:[]}]},{name:'아카이브',children:[]}]},{name:'개인 기록',children:[]}]};let path=[tree,tree.children[0],tree.children[0].children[0]];function paint(){q('nav').innerHTML=path.map((x,i)=>`<button data-i="${i}" ${i===path.length-1?'aria-current="page"':''}>${x.name}</button>${i<path.length-1?'<span aria-hidden="true">/</span>':''}`).join('');q('.places').innerHTML=path.at(-1).children.map((x,i)=>`<button class="place" data-child="${i}"><span aria-hidden="true">▱</span>${x.name}</button>`).join('')||'<p class="empty">빈 폴더</p>';qa('[data-i]').forEach(b=>b.onclick=()=>{path=path.slice(0,+b.dataset.i+1);paint()});qa('[data-child]').forEach(b=>b.onclick=()=>{path.push(path.at(-1).children[+b.dataset.child]);paint()})}root.onkeydown=e=>{if(e.key==='Escape'){path=[tree,tree.children[0],tree.children[0].children[0]];paint();q('[aria-current]').focus()}};paint();})();
</script>
