<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var list = root.querySelector('.u093-list'), bar = root.querySelector('.u093-bar i'), cnt = root.querySelector('.u093-c b'), win = root.querySelector('.u093-win');
  var od = root.querySelector('.u093-od>span'), again = root.querySelector('.u093-again'), cv = root.querySelector('.u093-cf'), live = root.querySelector('.u093-sr');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var TASKS = [['아침 스트레칭 10분', true], ['디자인 리뷰 메모 정리', true], ['물 2리터 마시기', false], ['내일 회의 안건 보내기', false]];
  var COLORS = ['#22C3A6', '#10B981', '#FFB547', '#FF6B8B', '#7B61FF', '#3D8BFF'];
  var items = TASKS.map(function (t) {
    var li = document.createElement('li');
    li.innerHTML = '<button class="u093-t" type="button" role="checkbox" aria-checked="' + t[1] + '"><span class="u093-bx" aria-hidden="true"><svg viewBox="0 0 26 26"><path d="M7.5 13.5l3.8 3.8L18.8 9"/></svg></span><span class="u093-tx"></span></button>';
    li.querySelector('.u093-tx').textContent = t[0];
    list.appendChild(li);
    return li.firstChild;
  });
  var ts = [], raf = 0;
  function later(fn, ms) { ts.push(setTimeout(function () { if (root.isConnected) fn(); }, ms)); }
  function done() { return items.filter(function (b) { return b.getAttribute('aria-checked') === 'true'; }).length; }
  function update() {
    var n = done();
    bar.style.setProperty('--p', n / items.length);
    if (cnt.textContent !== String(n)) {
      cnt.textContent = n;
      if (!rm) cnt.animate([{ transform: 'translateY(8px)', opacity: 0 }, { transform: 'none', opacity: 1 }], { duration: 300, easing: 'cubic-bezier(.2,.8,.2,1)' });
    }
    return n;
  }
  // 체크할 때마다 상자 둘레로 작은 점 여덟 개가 튑니다
  function pop(b) {
    if (rm) return;
    var rr = root.getBoundingClientRect(), s = root.offsetWidth / rr.width, br = b.querySelector('.u093-bx').getBoundingClientRect();
    var cx = (br.left + br.width / 2 - rr.left) * s, cy = (br.top + br.height / 2 - rr.top) * s;
    for (var i = 0; i < 8; i++) {
      var p = document.createElement('span'), a = i / 8 * Math.PI * 2 + 0.3, d = 20 + Math.random() * 8;
      p.className = 'u093-p'; p.style.left = cx + 'px'; p.style.top = cy + 'px'; p.style.background = COLORS[i % COLORS.length];
      root.appendChild(p);
      p.animate([{ transform: 'translate(0,0) scale(1)', opacity: 1 }, { transform: 'translate(' + Math.cos(a) * d + 'px,' + Math.sin(a) * d + 'px) scale(0)', opacity: 1 }], { duration: 520, easing: 'cubic-bezier(.2,.7,.3,1)' }).onfinish = function (e) { e.target.effect.target.remove(); };
    }
  }
  // 색종이는 배지 근처에서 위로 부채꼴로 터진 뒤 중력으로 떨어지고, 다 떨어지면 루프를 멈춥니다
  function confetti() {
    if (rm) return;
    var dpr = Math.min(2, window.devicePixelRatio || 1), W = root.offsetWidth, H = root.offsetHeight;
    cv.width = W * dpr; cv.height = H * dpr;
    var c = cv.getContext('2d'), ps = [];
    for (var i = 0; i < 90; i++) {
      var a = -Math.PI / 2 + (Math.random() - 0.5) * 1.9, v = 380 + Math.random() * 420;
      ps.push({ x: W / 2 + (Math.random() - 0.5) * 40, y: 180, vx: Math.cos(a) * v, vy: Math.sin(a) * v, r: Math.random() * 6, vr: (Math.random() - 0.5) * 14, w: 5 + Math.random() * 5, h: 8 + Math.random() * 6, col: COLORS[i % COLORS.length], t: 0, life: 1.6 + Math.random() * 0.9 });
    }
    var last = performance.now();
    cancelAnimationFrame(raf);
    (function f(now) {
      if (!root.isConnected) return;
      var dt = Math.min(0.033, (now - last) / 1000), alive = 0; last = now;
      c.setTransform(dpr, 0, 0, dpr, 0, 0); c.clearRect(0, 0, W, H);
      ps.forEach(function (p) {
        p.t += dt;
        if (p.t > p.life || p.y > H + 20) return;
        alive++;
        p.vy += 900 * dt; p.vx *= 0.985; p.vy *= 0.985;
        p.x += p.vx * dt; p.y += p.vy * dt; p.r += p.vr * dt;
        c.save(); c.translate(p.x, p.y); c.rotate(p.r);
        c.globalAlpha = Math.min(1, (p.life - p.t) / 0.4);
        c.scale(1, Math.cos(p.t * 9 + p.w));
        c.fillStyle = p.col; c.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
        c.restore();
      });
      if (alive) raf = requestAnimationFrame(f); else c.clearRect(0, 0, W, H);
    })(last);
  }
  function celebrate() {
    var had = list.contains(document.activeElement);
    root.classList.add('done');
    win.removeAttribute('aria-hidden');
    items.forEach(function (b) { b.tabIndex = -1; });
    od.style.setProperty('--v', 1);
    live.textContent = '오늘 할 일을 모두 끝냈어요. 연속 5일째예요';
    confetti();
    if (had) later(function () { again.focus({ preventScroll: true }); }, 400);
  }
  items.forEach(function (b) {
    b.addEventListener('click', function () {
      if (root.classList.contains('done')) return;
      var on = b.getAttribute('aria-checked') !== 'true';
      b.setAttribute('aria-checked', String(on));
      if (on) pop(b);
      var n = update();
      live.textContent = b.textContent + (on ? ' 완료' : ' 완료 취소') + ' · ' + items.length + '개 중 ' + n + '개 끝';
      if (n === items.length) later(celebrate, rm ? 0 : 450);
    });
  });
  again.addEventListener('click', function () {
    ts.forEach(clearTimeout); ts = [];
    root.classList.remove('done');
    win.setAttribute('aria-hidden', 'true');
    od.style.setProperty('--v', 0);
    items.forEach(function (b) { b.tabIndex = 0; });
    [3, 2].forEach(function (i, k) { later(function () { items[i].setAttribute('aria-checked', 'false'); update(); }, rm ? 0 : 350 + k * 140); });
    items[2].focus({ preventScroll: true });
  });
  update();
})();
</script>
