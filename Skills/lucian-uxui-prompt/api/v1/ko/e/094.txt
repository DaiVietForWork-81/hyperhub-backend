<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var sky = root.querySelector('.u094-sky'), orb = root.querySelector('.u094-orb'), tEl = root.querySelector('.u094-t'), cEl = root.querySelector('.u094-c'), now = root.querySelector('.u094-now');
  var strip = root.querySelector('.u094-strip'), labs = root.querySelector('.u094-labs'), trk = root.querySelector('.u094-trk'), kn = root.querySelector('.u094-kn');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var SKY = [[0, '#0B1026', '#1E2656'], [4.5, '#151B44', '#3B3F7A'], [6, '#F3A27E', '#6C7FD8'], [8, '#8ED0FF', '#4A92F0'], [12, '#5CB8FF', '#2F7BFF'], [14.5, '#7DA9D6', '#4E73B0'], [15.5, '#8C9AAD', '#56657A'], [17.4, '#8C9AAD', '#5C6A80'], [18.6, '#FF9F6E', '#7A5CD6'], [20, '#2A2E6E', '#141838'], [24, '#0B1026', '#1E2656']];
  var ICON = {
    sun: '<circle cx="11" cy="11" r="4.4" fill="#FFB020"/><path d="M11 2.5v2M11 17.5v2M2.5 11h2M17.5 11h2M5 5l1.4 1.4M15.6 15.6L17 17M5 17l1.4-1.4M15.6 6.4L17 5" stroke="#FFB020" stroke-width="1.8" stroke-linecap="round"/>',
    moon: '<path d="M14.5 15.8A6.6 6.6 0 0 1 8.2 4.4a6.6 6.6 0 1 0 8.6 9.2 6.6 6.6 0 0 1-2.3 2.2z" fill="#8B9BFF"/>',
    cloud: '<circle cx="8" cy="8" r="3.4" fill="#FFB020"/><path d="M7 18a3.6 3.6 0 0 1-.4-7.2 5 5 0 0 1 9.6.8A3.2 3.2 0 0 1 16.4 18z" fill="#A7B1C2"/>',
    rain: '<path d="M6.6 13a3.6 3.6 0 0 1-.4-7.2 5 5 0 0 1 9.6.8 3.2 3.2 0 0 1 .2 6.4z" fill="#A7B1C2"/><path d="M8 15.5l-1 3M11.5 15.5l-1 3M15 15.5l-1 3" stroke="#3D8BFF" stroke-width="1.8" stroke-linecap="round"/>'
  };
  function cond(h) { return h < 6 || h >= 19.5 ? ['맑은 밤', 'moon', .12] : h < 12 ? ['맑음', 'sun', .1] : h < 15 ? ['구름 조금', 'cloud', .55] : h < 17.5 ? ['소나기', 'rain', 1] : ['갬', 'cloud', .35]; }
  function temp(h) { return Math.round(18 + 5 * Math.sin((h - 9) / 24 * Math.PI * 2) - (h >= 15 && h < 17.5 ? 2 : 0)); }
  function hx(c) { return [parseInt(c.slice(1, 3), 16), parseInt(c.slice(3, 5), 16), parseInt(c.slice(5, 7), 16)]; }
  function mix(a, b, t) { a = hx(a); b = hx(b); return 'rgb(' + a.map(function (v, i) { return Math.round(v + (b[i] - v) * t); }).join(',') + ')'; }
  function label(h) { var hh = Math.floor(h), mm = Math.round((h - hh) * 60) >= 30 ? '30' : '00'; return (hh < 12 ? '오전 ' : '오후 ') + ((hh % 12) || 12) + ':' + mm; }
  for (var i = 0; i < 18; i++) {
    var s = document.createElement('i'); s.style.left = (8 + Math.random() * 84) + '%'; s.style.top = (6 + Math.random() * 60) + '%'; s.style.animationDelay = (Math.random() * 2.6) + 's';
    root.querySelector('.u094-stars').appendChild(s);
  }
  for (i = 0; i < 22; i++) {
    var d = document.createElement('i'); d.style.left = (Math.random() * 120) + '%'; d.style.animationDelay = (-Math.random() * 0.8) + 's'; d.style.animationDuration = (0.6 + Math.random() * 0.4) + 's';
    root.querySelector('.u094-drops').appendChild(d);
  }
  var LB = [3, 6, 9, 12, 15, 18, 21].map(function (h) {
    var b = document.createElement('button');
    b.type = 'button'; b.className = 'u094-lb'; b.tabIndex = -1; b.style.left = (h / 24 * 100) + '%';
    b.innerHTML = '<small>' + h + '시</small><svg viewBox="0 0 22 22" aria-hidden="true">' + ICON[cond(h)[1]] + '</svg><b>' + temp(h) + '°</b>';
    b.setAttribute('aria-label', h + '시 ' + temp(h) + '도 ' + cond(h)[0]);
    b.addEventListener('click', function (e) { e.stopPropagation(); glide(h); });
    b.addEventListener('pointerdown', function (e) { e.stopPropagation(); });
    labs.appendChild(b);
    return [h, b];
  });
  var hour = 9, shownT = null, tw = 0;
  // 시각 하나로 하늘 색, 해·달 위치, 구름·비·별, 온도를 모두 계산합니다
  function set(h) {
    hour = (h + 24) % 24;
    var k = 0; while (SKY[k + 1][0] <= hour) k++;
    var a = SKY[k], b = SKY[k + 1], t = (hour - a[0]) / (b[0] - a[0]);
    sky.style.setProperty('--s1', mix(a[1], b[1], t)); sky.style.setProperty('--s2', mix(a[2], b[2], t));
    var day = hour >= 6 && hour < 18, p = day ? (hour - 6) / 12 : ((hour - 18 + 24) % 24) / 12;
    var x = (1 - p) * (1 - p) * 120 + 2 * (1 - p) * p * 225 + p * p * 330, y = (1 - p) * (1 - p) * 206 + 2 * (1 - p) * p * -118 + p * p * 206;
    orb.className = 'u094-orb ' + (day ? 'sun' : 'moon');
    orb.style.setProperty('--ox', x.toFixed(1) + 'px'); orb.style.setProperty('--oy', y.toFixed(1) + 'px');
    orb.style.opacity = Math.min(1, Math.sin(Math.PI * p) * 3.5).toFixed(2);
    var c = cond(hour);
    sky.style.setProperty('--cloud', c[2]);
    sky.style.setProperty('--night', day ? Math.max(0, 1 - (Math.min(hour - 6, 18 - hour)) * 1.5).toFixed(2) : 1);
    root.classList.toggle('rain', c[1] === 'rain');
    var T = temp(hour);
    if (T !== shownT) {
      if (shownT !== null && !rm) tEl.animate([{ transform: 'translateY(' + (T > shownT ? 10 : -10) + 'px)', opacity: 0.2 }, { transform: 'none', opacity: 1 }], { duration: 260, easing: 'cubic-bezier(.2,.8,.2,1)' });
      shownT = T; tEl.textContent = T + '°';
    }
    cEl.textContent = c[0];
    now.textContent = label(hour);
    trk.style.setProperty('--x', (hour / 24 * 100).toFixed(2) + '%');
    kn.classList.toggle('moon', !day);
    var near = LB.reduce(function (m, l) { return Math.abs(l[0] - hour) < Math.abs(m[0] - hour) ? l : m; });
    LB.forEach(function (l) { l[1].classList.toggle('on', l === near && Math.abs(l[0] - hour) <= 1.5); });
    strip.setAttribute('aria-valuenow', hour.toFixed(1));
    strip.setAttribute('aria-valuetext', label(hour) + ', ' + T + '도, ' + c[0]);
  }
  function glide(to) {
    cancelAnimationFrame(tw);
    var from = hour, t0 = performance.now(), dur = rm ? 1 : 700;
    (function f(n) {
      if (!root.isConnected) return;
      var k = Math.min(1, (n - t0) / dur), e = k < .5 ? 4 * k * k * k : 1 - Math.pow(-2 * k + 2, 3) / 2;
      set(from + (to - from) * e);
      if (k < 1) tw = requestAnimationFrame(f);
    })(t0);
  }
  function at(e) { var r = trk.getBoundingClientRect(); return Math.max(0, Math.min(23.5, (e.clientX - r.left) / r.width * 24)); }
  var down = false;
  strip.addEventListener('pointerdown', function (e) { if (e.button !== 0) return; down = true; cancelAnimationFrame(tw); strip.setPointerCapture(e.pointerId); strip.classList.add('drag'); set(at(e)); });
  strip.addEventListener('pointermove', function (e) { if (down) set(at(e)); });
  function up() { if (!down) return; down = false; strip.classList.remove('drag'); glide(Math.round(hour * 2) / 2); }
  strip.addEventListener('pointerup', up); strip.addEventListener('pointercancel', up);
  strip.addEventListener('keydown', function (e) {
    var to = null, r = Math.round(hour);
    if (e.key === 'ArrowRight' || e.key === 'ArrowUp') to = Math.min(23.5, r + 1);
    else if (e.key === 'ArrowLeft' || e.key === 'ArrowDown') to = Math.max(0, r - 1);
    else if (e.key === 'Home') to = 0;
    else if (e.key === 'End') to = 23.5;
    if (to === null) return;
    e.preventDefault();
    glide(to);
  });
  set(9);
  if (rm) set(14); else setTimeout(function () { if (root.isConnected) glide(14); }, 400);
})();
</script>
