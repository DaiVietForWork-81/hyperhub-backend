What it does: 도형을 옮길 때 가운데 가이드에 달라붙는 정렬 캔버스입니다. 중앙 가까이 가면 가이드선이 나타나며 정확히 맞춰지고, Enter로 바로 중앙에 둘 수도 있습니다.

<div class="ux u125"><div class="canvas" aria-label="스냅 정렬 영역"><span class="vline"></span><span class="hline"></span><button class="object" aria-label="도형 위치 X 30%, Y 40%, 방향키로 이동, Enter로 중앙 정렬"></button></div><output class="sr" aria-live="polite"></output></div>
<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];let x=30,y=40;const snap=v=>Math.abs(v-50)<=3?50:v;function paint(){x=Math.max(15,Math.min(85,snap(x)));y=Math.max(15,Math.min(85,snap(y)));root.style.setProperty('--x',x+'%');root.style.setProperty('--y',y+'%');q('.canvas').classList.toggle('snapx',x===50);q('.canvas').classList.toggle('snapy',y===50);const text=`X ${Math.round(x)}%, Y ${Math.round(y)}%${x===50&&y===50?', 중앙 정렬':''}`;q('output').textContent=text;q('.object').setAttribute('aria-label','도형 위치 '+text+', 방향키로 이동, Enter로 중앙 정렬')}q('.object').onkeydown=e=>{let d={ArrowLeft:[-5,0],ArrowRight:[5,0],ArrowUp:[0,-5],ArrowDown:[0,5]}[e.key];if(d){e.preventDefault();x+=d[0];y+=d[1];paint()}if(e.key==='Enter'){e.preventDefault();x=50;y=50;paint()}if(e.key==='Escape'){x=30;y=40;paint()}};q('.object').onpointerdown=e=>{e.currentTarget.setPointerCapture(e.pointerId);e.currentTarget.classList.add('drag')};q('.object').onpointermove=e=>{if(e.currentTarget.hasPointerCapture(e.pointerId)){const r=q('.canvas').getBoundingClientRect();x=(e.clientX-r.left)/r.width*100;y=(e.clientY-r.top)/r.height*100;paint()}};q('.object').onpointerup=e=>e.currentTarget.classList.remove('drag');q('.object').onpointercancel=e=>e.currentTarget.classList.remove('drag');paint();})();
</script>
