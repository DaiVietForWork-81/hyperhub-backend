<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var fab = root.querySelector('.u011-fab');
  var items = Array.prototype.slice.call(root.querySelectorAll('.u011-it'));
  var note = root.querySelector('.u011-note');
  var noteT = root.querySelector('.u011-note span');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var t = 0;
  function setOpen(v) {
    root.classList.toggle('is-open', v);
    fab.setAttribute('aria-expanded', String(v));
    fab.setAttribute('aria-label', v ? '메뉴 닫기' : '새로 만들기');
    items.forEach(function (it) { it.tabIndex = v ? 0 : -1; });
    if (v) note.classList.remove('is-on');
  }
  setOpen(false);
  fab.addEventListener('click', function () { setOpen(!root.classList.contains('is-open')); });
  items.forEach(function (it, i) {
    it.addEventListener('click', function () {
      it.classList.remove('is-picked'); void it.offsetWidth; it.classList.add('is-picked');
      clearTimeout(t);
      t = setTimeout(function () {
        if (!root.isConnected) return;
        setOpen(false);
        noteT.textContent = it.dataset.done;
        note.classList.add('is-on');
        fab.focus({ preventScroll: true });
        t = setTimeout(function () { if (root.isConnected) note.classList.remove('is-on'); }, 1800);
      }, rm ? 0 : 260);
    });
    it.addEventListener('keydown', function (e) {
      var d = e.key === 'ArrowRight' || e.key === 'ArrowDown' ? 1 : e.key === 'ArrowLeft' || e.key === 'ArrowUp' ? -1 : 0;
      if (d) { e.preventDefault(); items[(i + d + items.length) % items.length].focus(); }
    });
  });
  root.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && root.classList.contains('is-open')) { e.preventDefault(); setOpen(false); fab.focus(); }
  });
  document.addEventListener('pointerdown', function (e) {
    if (root.isConnected && root.classList.contains('is-open') && !root.contains(e.target)) setOpen(false);
  });
})();
</script>
