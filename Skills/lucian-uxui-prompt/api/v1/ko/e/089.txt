<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var num = root.querySelector('.u089-num'), cols = root.querySelectorAll('.u089-col'), cp = root.querySelector('.u089-cp'), nd = root.querySelector('.u089-nd');
  var path = root.querySelector('.u089-url b'), title = root.querySelector('.u089-t'), sub = root.querySelector('.u089-s'), sug = root.querySelector('.u089-sug'), re = root.querySelector('.u089-re');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var P = [['/projects', '프로젝트', 0.92], ['/profile', '내 프로필', 0.58], ['/pricing', '요금제', 0.41]];
  var ARROW = '<path d="M3 8h10M9 4l4 4-4 4"/>', CHECK = '<path d="M3.2 8.4l3 3L12.8 4.8"/>';
  var rows = P.map(function (p, i) {
    var li = document.createElement('li');
    li.innerHTML = '<button class="u089-row" type="button"><code></code><small></small><span class="u089-m" aria-hidden="true"><i style="--m:' + p[2] + '"></i></span><svg viewBox="0 0 16 16" aria-hidden="true">' + ARROW + '</svg></button>';
    var b = li.firstChild;
    b.querySelector('code').textContent = p[0]; b.querySelector('small').textContent = p[1];
    b.setAttribute('aria-label', p[0] + ' ' + p[1] + ' 페이지로 가기, 일치도 ' + Math.round(p[2] * 100) + '%');
    b.addEventListener('click', function () { go(i); });
    b.addEventListener('focus', function () { aimAt(b); });
    sug.appendChild(li);
    return b;
  });
  // 바늘은 스프링으로 움직여서 방향이 크게 바뀌면 살짝 지나쳤다가 돌아옵니다
  var a = 0, v = 0, tgt = 0, raf = 0, last = 0, ts = [], busy = false;
  function step(now) {
    raf = 0;
    if (!root.isConnected) return;
    var dt = Math.min(0.032, (now - last) / 1000); last = now;
    if (rm) { a = tgt; v = 0; } else { v += (220 * (tgt - a) - 17 * v) * dt; a += v * dt; }
    nd.style.transform = 'rotate(' + a.toFixed(2) + 'deg)';
    if (Math.abs(v) > 0.05 || Math.abs(tgt - a) > 0.05) raf = requestAnimationFrame(step);
  }
  function kick() { if (!raf) { last = performance.now(); raf = requestAnimationFrame(step); } }
  function aim(deg) { tgt = a + ((((deg - a) % 360) + 540) % 360) - 180; kick(); }
  function toward(x, y) {
    var r = cp.getBoundingClientRect(), dx = x - (r.left + r.width / 2), dy = y - (r.top + r.height / 2);
    aim(Math.atan2(dx, -dy) * 180 / Math.PI);
    num.style.setProperty('--r', Math.max(-1, Math.min(1, dx / (root.getBoundingClientRect().width / 2))).toFixed(3));
  }
  function aimAt(el) { if (busy || root.classList.contains('found')) return; root.classList.add('trk'); var r = el.getBoundingClientRect(); toward(r.left + r.width / 2, r.top + r.height / 2); }
  root.addEventListener('pointermove', function (e) { if (busy || root.classList.contains('found')) return; root.classList.add('trk'); toward(e.clientX, e.clientY); });
  root.addEventListener('pointerleave', function () { if (root.contains(document.activeElement)) return; root.classList.remove('trk'); num.style.setProperty('--r', 0); aim(0); });
  root.addEventListener('focusout', function (e) { if (!root.contains(e.relatedTarget) && !root.matches(':hover')) { root.classList.remove('trk'); num.style.setProperty('--r', 0); aim(0); } });
  function later(fn, ms) { ts.push(setTimeout(function () { if (root.isConnected) fn(); }, ms)); }
  function swapText(t, s) {
    root.classList.add('swap');
    later(function () { title.textContent = t; sub.textContent = s; root.classList.remove('swap'); }, rm ? 0 : 250);
  }
  function digits(x, y, label) { cols[0].style.setProperty('--v', x); cols[1].style.setProperty('--v', y); num.setAttribute('aria-label', label); }
  // 오타 글자를 한 자씩 지운 뒤 맞는 주소를 다시 입력하고, 숫자가 굴러 200이 됩니다
  function go(i) {
    if (busy || root.classList.contains('found')) return;
    busy = true;
    rows.forEach(function (b, k) { b.disabled = k !== i; });
    rows[i].classList.add('on');
    root.classList.add('fixing');
    var from = path.textContent, to = P[i][0], k = from.length;
    (function del() {
      if (k > 1) { path.textContent = from.slice(0, --k); return later(del, rm ? 0 : 34); }
      var j = 1;
      (function type() {
        path.textContent = to.slice(0, ++j);
        if (j < to.length) return later(type, rm ? 0 : 55);
        root.classList.remove('fixing', 'trk');
        root.classList.add('found');
        num.style.setProperty('--r', 0);
        digits(2, 0, '200 정상');
        tgt = a - (((a % 360) + 360) % 360) + 720; kick();
        rows[i].querySelector('svg').innerHTML = CHECK;
        rows[i].disabled = false;
        swapText('찾았어요', '‘' + P[i][1] + '’ 페이지를 열었어요');
        busy = false;
      })();
    })();
  }
  re.addEventListener('click', function () {
    ts.forEach(clearTimeout); ts = [];
    root.classList.remove('found', 'fixing');
    path.textContent = '/projcts';
    digits(4, 4, '404 오류');
    rows.forEach(function (b) { b.disabled = false; b.classList.remove('on'); b.querySelector('svg').innerHTML = ARROW; });
    swapText('길을 잃은 것 같아요', '주소에 오타가 있거나 페이지가 옮겨졌어요');
    rows[0].focus({ preventScroll: true });
  });
})();
</script>
