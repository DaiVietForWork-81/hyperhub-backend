<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var sws = Array.prototype.slice.call(root.querySelectorAll('.u023-sw'));
  var more = root.querySelector('.u023-more');
  var mail = root.querySelector('.u023-mail');
  var opts = Array.prototype.slice.call(root.querySelectorAll('.u023-opt'));
  var pill = root.querySelector('.u023-pill');
  var badge = root.querySelector('.u023-badge');
  var all = root.querySelector('.u023-all');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var freq = 1;
  function setSw(sw, on) {
    sw.setAttribute('aria-checked', String(on));
    sw.closest('.u023-item').classList.toggle('is-off', !on);
    if (sw.hasAttribute('data-more')) {
      more.classList.toggle('is-open', on);
      mail.textContent = on ? opts[freq].textContent + ' 아침 8시에 보내요' : '꺼져 있어요';
      opts.forEach(function (o) { o.tabIndex = on && o.getAttribute('aria-checked') === 'true' ? 0 : -1; });
    }
    var n = sws.filter(function (s) { return s.getAttribute('aria-checked') === 'true'; }).length;
    badge.textContent = n ? n + '개 켜짐' : '모두 꺼짐';
  }
  sws.forEach(function (sw) {
    setSw(sw, sw.getAttribute('aria-checked') === 'true');
    sw.addEventListener('click', function () { setSw(sw, sw.getAttribute('aria-checked') !== 'true'); });
  });
  function pick(i, focus) {
    freq = i;
    opts.forEach(function (o, k) { o.setAttribute('aria-checked', String(k === i)); o.tabIndex = k === i ? 0 : -1; });
    pill.style.setProperty('--i', i);
    mail.textContent = opts[i].textContent + ' 아침 8시에 보내요';
    if (focus) opts[i].focus();
  }
  opts.forEach(function (o, i) {
    o.addEventListener('click', function () { pick(i); });
    o.addEventListener('keydown', function (e) {
      var d = e.key === 'ArrowRight' ? 1 : e.key === 'ArrowLeft' ? -1 : 0;
      if (d) { e.preventDefault(); pick((i + d + 3) % 3, true); }
    });
  });
  // 모두 끄기: 위에서부터 0.12초 간격으로 차례차례 끕니다
  all.addEventListener('click', function () {
    sws.forEach(function (sw, i) {
      setTimeout(function () { if (root.isConnected) setSw(sw, false); }, rm ? 0 : i * 120);
    });
  });
})();
</script>
