<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var track = root.querySelector('.u017-track');
  var thumb = root.querySelector('.u017-thumb');
  var bub = root.querySelector('.u017-bub');
  var out = root.querySelector('.u017-val');
  var ticks = Array.prototype.slice.call(root.querySelectorAll('.u017-ticks span'));
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var MAX = 1000000, STEP = 10000;
  var v = 450000, lastX = 0, lastT = 0, still = 0;
  function render() {
    var p = v / MAX;
    root.style.setProperty('--v', p.toFixed(4));
    out.textContent = v.toLocaleString('ko-KR') + '원';
    thumb.setAttribute('aria-valuenow', String(v));
    thumb.setAttribute('aria-valuetext', (v / 10000) + '만 원');
    ticks.forEach(function (t, i) { t.classList.toggle('on', i / 4 <= p + 0.001); });
  }
  // 움직이는 속도에 비례해 말풍선이 반대쪽으로 기울고, 멈추면 0도로 돌아옵니다
  function tilt(deg) {
    if (rm) return;
    bub.style.setProperty('--tilt', Math.max(-22, Math.min(22, deg)).toFixed(1) + 'deg');
    clearTimeout(still);
    still = setTimeout(function () { bub.style.setProperty('--tilt', '0deg'); }, 90);
  }
  function fromX(x) {
    var r = track.getBoundingClientRect();
    var p = Math.max(0, Math.min(1, (x - r.left) / r.width));
    v = Math.round(p * MAX / STEP) * STEP;
    render();
  }
  track.addEventListener('pointerdown', function (e) {
    if (e.button !== 0) return;
    track.setPointerCapture(e.pointerId);
    root.classList.add('is-drag');
    lastX = e.clientX; lastT = performance.now();
    fromX(e.clientX);
    thumb.focus({ preventScroll: true });
  });
  track.addEventListener('pointermove', function (e) {
    if (!track.hasPointerCapture(e.pointerId)) return;
    var now = performance.now();
    var vel = (e.clientX - lastX) / Math.max(8, now - lastT);
    lastX = e.clientX; lastT = now;
    tilt(-vel * 14);
    fromX(e.clientX);
  });
  function end() { root.classList.remove('is-drag'); bub.style.setProperty('--tilt', '0deg'); }
  track.addEventListener('pointerup', end);
  track.addEventListener('pointercancel', end);
  thumb.addEventListener('keydown', function (e) {
    var d = { ArrowRight: STEP, ArrowUp: STEP, ArrowLeft: -STEP, ArrowDown: -STEP, PageUp: 100000, PageDown: -100000 }[e.key];
    if (e.key === 'Home') d = -MAX; if (e.key === 'End') d = MAX;
    if (!d) return;
    e.preventDefault();
    v = Math.max(0, Math.min(MAX, v + d));
    tilt(d > 0 ? -9 : 9);
    render();
  });
  render();
})();
</script>
