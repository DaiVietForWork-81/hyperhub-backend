<script>
(()=>{const root=document.currentScript.previousElementSibling,q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];function lum(hex){const rgb=[1,3,5].map(i=>parseInt(hex.slice(i,i+2),16)/255).map(c=>c<=.04045?c/12.92:Math.pow((c+.055)/1.055,2.4));return rgb[0]*.2126+rgb[1]*.7152+rgb[2]*.0722}function paint(){let valid=true;qa('input').forEach(el=>{let ok=/^#[a-f0-9]{6}$/i.test(el.value);el.setAttribute('aria-invalid',!ok);valid=valid&&ok});if(!valid)return;const f=q('.fg').value,b=q('.bg').value,l1=lum(f),l2=lum(b),ratio=(Math.max(l1,l2)+.05)/(Math.min(l1,l2)+.05);q('.sample').style.color=f;q('.sample').style.background=b;q('.number').textContent=ratio.toFixed(2)+' : 1';q('.grade').textContent=ratio>=4.5?'✓ 일반 AA':'× 일반 AA 미달'}qa('input').forEach(x=>x.oninput=paint);paint();})();
</script>
