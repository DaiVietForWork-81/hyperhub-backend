<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var vp = root.querySelector('.u085-vp'), list = root.querySelector('.u085-list'), msg = root.querySelector('.u085-msg'), rb = root.querySelector('.u085-rb');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var POOL = [
    ['윤하린', '봄 캠페인 시안 v3를 올렸어요', 'linear-gradient(140deg,#22C3A6,#0EA5E9)'],
    ['한도윤', '버튼 전환 속도 실험 결과 공유', 'linear-gradient(140deg,#8B6CFF,#5B8CFF)'],
    ['정미나', 'QA 체크리스트가 업데이트됐어요', 'linear-gradient(140deg,#FF7A93,#B04FD0)'],
    ['Lumen 봇', '오늘 배포 3건이 모두 성공했어요', 'linear-gradient(140deg,#FFB547,#FF7A45)'],
    ['박지호', '다음 주 밋업 장소를 정했어요', 'linear-gradient(140deg,#3D8BFF,#22C3A6)'],
    ['김서연', '디자인 시스템 2.4 릴리스 노트', 'linear-gradient(140deg,#B04FD0,#FF6B8B)'],
    ['최우진', '아이콘 세트 48개를 추가했어요', 'linear-gradient(140deg,#0EA5E9,#7B61FF)']
  ];
  var next = 0, mins = 0, busy = false;
  function item(d, ago, fresh, i) {
    var li = document.createElement('li');
    li.className = 'u085-it' + (fresh ? ' fresh' : '');
    li.style.setProperty('--i', i || 0);
    li.innerHTML = '<i style="--g:' + d[2] + '" aria-hidden="true"></i><div><b></b><span></span></div>';
    li.querySelector('b').textContent = d[0] + ' · ' + d[1];
    li.querySelector('span').textContent = ago;
    return li;
  }
  for (var k = 0; k < 5; k++) list.appendChild(item(POOL[next++], (k + 1) * 12 + '분 전'));
  function set(p) { root.style.setProperty('--pull', p.toFixed(1)); root.classList.toggle('is-ready', p >= 70); msg.textContent = p >= 70 ? '놓으면 새로고침' : '아래로 당겨 새로고침'; }
  // 새 글은 맨 위로 밀려 들어오고, 잠깐 강조된 뒤 원래 색으로 돌아옵니다
  function refresh() {
    if (busy) return;
    busy = true;
    root.classList.add('rel', 'is-load');
    set(70);
    msg.textContent = '새 소식을 불러오는 중';
    setTimeout(function () {
      if (!root.isConnected) return;
      var add = 1 + (mins % 2);
      for (var i = add - 1; i >= 0; i--) { list.insertBefore(item(POOL[next % POOL.length], '방금', true, i), list.firstChild); next++; }
      while (list.children.length > 6) list.lastChild.remove();
      mins++;
      root.classList.remove('is-load');
      set(0);
      setTimeout(function () { list.querySelectorAll('.fresh').forEach(function (x) { x.classList.remove('fresh'); }); busy = false; }, 1600);
    }, rm ? 300 : 1300);
  }
  // 당긴 거리에 저항을 줘서 갈수록 무겁게 따라오게 합니다
  var sy = 0, down = false, pull = 0;
  vp.addEventListener('pointerdown', function (e) { if (busy || e.button !== 0) return; down = true; sy = e.clientY; pull = 0; vp.setPointerCapture(e.pointerId); vp.classList.add('drag'); root.classList.remove('rel'); });
  vp.addEventListener('pointermove', function (e) {
    if (!down) return;
    var k = vp.offsetHeight / vp.getBoundingClientRect().height, d = Math.max(0, (e.clientY - sy) * k);
    pull = 120 * (1 - Math.exp(-d / 140));
    set(pull);
  });
  function end() { if (!down) return; down = false; vp.classList.remove('drag'); if (pull >= 70) refresh(); else { root.classList.add('rel'); set(0); } }
  vp.addEventListener('pointerup', end); vp.addEventListener('pointercancel', end);
  vp.addEventListener('keydown', function (e) { if (e.key === 'r' || e.key === 'R') { e.preventDefault(); refresh(); } });
  rb.addEventListener('click', refresh);
  set(0);
})();
</script>
