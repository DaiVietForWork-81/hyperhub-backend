<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var its = Array.prototype.slice.call(root.querySelectorAll('.u072-it')), n = root.querySelector('.u072-n');
  var calm = root.querySelector('.u072-calm'), again = root.querySelector('.u072-again'), more = root.querySelector('.u072-more'), log = root.querySelector('.u072-log');
  function count() {
    var k = its.filter(function (it) { return !it.classList.contains('gone'); }).length;
    n.textContent = k ? '알림 ' + k + '개' : '';
    root.classList.toggle('is-clear', !k);
    calm.setAttribute('aria-hidden', String(!!k));
    again.tabIndex = k ? -1 : 0;
    return k;
  }
  // 닫으면 먼저 옆으로 밀려나고, 이어서 높이가 접히며 아래 줄이 올라옵니다
  its.forEach(function (it, i) {
    it.querySelector('.u072-x').addEventListener('click', function () {
      it.classList.add('gone'); it.inert = true;
      var rest = its.filter(function (x) { return !x.classList.contains('gone'); });
      var next = rest.filter(function (x) { return its.indexOf(x) > i; })[0] || rest[rest.length - 1];
      if (count() && next) next.querySelector('.u072-x').focus({ preventScroll: true });
      else setTimeout(function () { if (root.isConnected) again.focus({ preventScroll: true }); }, 500);
    });
  });
  more.addEventListener('click', function () {
    var on = more.getAttribute('aria-expanded') !== 'true';
    more.setAttribute('aria-expanded', String(on));
    more.firstChild.textContent = on ? '접기 ' : '자세히 ';
    log.classList.toggle('on', on);
  });
  again.addEventListener('click', function () {
    its.forEach(function (it, i) { setTimeout(function () { if (!root.isConnected) return; it.classList.remove('gone'); it.inert = false; if (i === its.length - 1) count(); }, i * 90); });
    root.classList.remove('is-clear');
    setTimeout(function () { if (root.isConnected) its[0].querySelector('.u072-x').focus({ preventScroll: true }); }, 200);
  });
  count();
})();
</script>
