<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var uid = 'u098' + Math.random().toString(36).slice(2, 7);
  var num = root.querySelector('.u098-num'), tt = root.querySelector('.u098-tt'), pp = root.querySelector('.u098-pp'), bd = root.querySelector('.u098-bd'), bdv = bd.querySelector('b');
  var g = root.querySelector('.u098-g'), ln = root.querySelector('.u098-ln'), ar = root.querySelector('.u098-ar'), hd = root.querySelector('.u098-hd'), tabs = root.querySelector('.u098-tabs');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  root.querySelector('.u098-gd').id = uid; ar.setAttribute('fill', 'url(#' + uid + ')');
  var PLAY = '<path d="M3.5 2.2v9.6L11.6 7z"/>', PAUSE = '<rect x="3" y="2.5" width="3" height="9" rx="1"/><rect x="8" y="2.5" width="3" height="9" rx="1"/>';
  var N = 24, W = 340, H = 128, STEP = (W - 16) / (N - 1);
  function walk(seed, a, vol, n) { var s = seed, v = a, out = []; for (var i = 0; i < n; i++) { s = (s * 9301 + 49297) % 233280; v += (s / 233280 - 0.48) * vol; out.push(v); } return out; }
  var M = [
    { name: '방문자', title: '실시간 방문자', v: walk(3, 12400, 90, N), vol: 70, fmt: function (v) { return Math.round(v).toLocaleString('ko-KR'); }, tail: '', base: 12090 },
    { name: '전환율', title: '실시간 전환율', v: walk(8, 3.6, 0.05, N), vol: 0.04, fmt: function (v) { return v.toFixed(2); }, tail: '%', base: 3.71 },
    { name: '매출', title: '오늘 매출', v: walk(5, 8200, 40, N), vol: 30, fmt: function (v) { return Math.round(v).toLocaleString('ko-KR'); }, tail: '만', head: '₩', base: 7930 }
  ];
  var k = 0, ys = null, paused = false, seen = true, timer = 0, busy = false, tw = 0, shown = '';
  var tb = M.map(function (m, i) {
    var b = document.createElement('button');
    b.type = 'button'; b.className = 'u098-tb'; b.setAttribute('aria-pressed', String(i === 0));
    b.innerHTML = '<small></small><b></b><em></em>';
    b.querySelector('small').textContent = m.name;
    b.addEventListener('click', function () { if (i !== k) pick(i); });
    tabs.appendChild(b);
    return b;
  });
  function last(m) { return m.v[m.v.length - 1]; }
  function text(m) { return (m.head || '') + m.fmt(last(m)) + m.tail; }
  // 숫자 칸은 0~9를 세로로 쌓은 기둥이라, 바뀐 자리만 그 높이만큼 위아래로 움직입니다
  function digits(str, fresh) {
    var cols = num.querySelectorAll('.u098-col');
    var same = !fresh && shown.replace(/\d/g, '0') === str.replace(/\d/g, '0');
    if (!same) {
      num.innerHTML = '';
      str.split('').forEach(function (ch, i) {
        var s = document.createElement('span');
        if (/\d/.test(ch)) {
          s.className = 'u098-col'; s.style.setProperty('--d', 0); s.style.setProperty('--i', i);
          s.innerHTML = '<span>0</span><span>1</span><span>2</span><span>3</span><span>4</span><span>5</span><span>6</span><span>7</span><span>8</span><span>9</span>';
        } else { s.className = 'u098-sym' + (ch === '만' || ch === '%' ? ' sm' : ''); s.textContent = ch; }
        num.appendChild(s);
      });
      void num.offsetWidth;
      cols = num.querySelectorAll('.u098-col');
    }
    var j = 0;
    str.split('').forEach(function (ch) { if (/\d/.test(ch)) cols[j++].style.setProperty('--d', ch); });
    shown = str;
    num.setAttribute('aria-label', M[k].title + ' ' + str);
  }
  function domain(v) { var mn = Math.min.apply(null, v), mx = Math.max.apply(null, v), p = (mx - mn) * 0.35 || 1; return [mn - p, mx + p]; }
  var dom = domain(M[0].v);
  function Y(v) { return 8 + (1 - (v - dom[0]) / (dom[1] - dom[0])) * (H - 16); }
  function path(y) { return y.map(function (v, i) { return (i ? 'L' : 'M') + (i * STEP).toFixed(1) + ' ' + v.toFixed(1); }).join(''); }
  function paint(y) {
    var p = path(y), n = y.length - 1;
    ln.setAttribute('d', p); ar.setAttribute('d', p + 'L' + (n * STEP).toFixed(1) + ' ' + (H + 4) + 'L0 ' + (H + 4) + 'Z');
    hd.setAttribute('cx', (n * STEP).toFixed(1)); hd.setAttribute('cy', y[n].toFixed(1));
  }
  function badge(m) {
    var r = (last(m) - m.base) / m.base, up = r >= 0;
    bd.classList.toggle('dn', !up); bd.style.setProperty('--c', up ? 'var(--ux-mint)' : 'var(--ux-danger)');
    bdv.textContent = Math.abs(r * 100).toFixed(1) + '%';
    tb.forEach(function (b, i) {
      var mm = M[i], rr = (last(mm) - mm.base) / mm.base;
      b.querySelector('b').textContent = text(mm);
      var em = b.querySelector('em'); em.textContent = (rr >= 0 ? '+' : '−') + Math.abs(rr * 100).toFixed(1) + '%'; em.classList.toggle('dn', rr < 0);
    });
  }
  // 새 값이 오면 점 하나를 오른쪽 밖에 붙여 그린 뒤, 묶음 전체를 한 칸 왼쪽으로 흘려보냅니다
  function tick() {
    timer = 0;
    if (!root.isConnected) return;
    if (paused || !seen || busy) return schedule();
    var m = M[k], prev = last(m), mid = (dom[0] + dom[1]) / 2;
    var nv = Math.max(dom[0] + (dom[1] - dom[0]) * 0.12, Math.min(dom[1] - (dom[1] - dom[0]) * 0.12, prev + (Math.random() - 0.5) * m.vol * 2 + (mid - prev) * 0.15));
    M.forEach(function (mm, i) { if (i !== k) { mm.v.push(last(mm) + (Math.random() - 0.5) * mm.vol); mm.v.shift(); } });
    m.v.push(nv);
    ys = ys.concat([Y(nv)]);
    paint(ys);
    digits(text(m)); badge(m);
    if (rm) { m.v.shift(); ys.shift(); paint(ys); return schedule(); }
    busy = true;
    var t0 = performance.now();
    (function f(n) {
      if (!root.isConnected) return;
      var t = Math.min(1, (n - t0) / 700), e = 1 - Math.pow(1 - t, 3);
      g.setAttribute('transform', 'translate(' + (-STEP * e).toFixed(2) + ' 0)');
      if (t < 1) return requestAnimationFrame(f);
      m.v.shift(); ys.shift(); g.removeAttribute('transform'); paint(ys); busy = false;
    })(t0);
    schedule();
  }
  function schedule() { clearTimeout(timer); timer = setTimeout(tick, 2400); }
  // 지표를 바꾸면 숫자는 새로 세운 기둥이 0에서부터 굴러 올라가고, 선은 같은 칸 수로 모양만 바뀝니다
  function pick(i) {
    k = i;
    tb.forEach(function (b, j) { b.setAttribute('aria-pressed', String(j === i)); });
    tt.textContent = M[i].title;
    dom = domain(M[i].v);
    var to = M[i].v.map(Y), from = ys.slice(0, to.length), t0 = performance.now();
    cancelAnimationFrame(tw);
    (function f(n) {
      if (!root.isConnected) return;
      var t = rm ? 1 : Math.min(1, (n - t0) / 600), e = t < .5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
      ys = from.map(function (y, j) { return y + (to[j] - y) * e; }); paint(ys);
      if (t < 1) tw = requestAnimationFrame(f);
    })(t0);
    digits(text(M[i]), true); badge(M[i]);
    schedule();
  }
  pp.addEventListener('click', function () {
    paused = !paused;
    root.classList.toggle('paused', paused);
    pp.setAttribute('aria-pressed', String(paused));
    pp.setAttribute('aria-label', paused ? '실시간 갱신 다시 시작' : '실시간 갱신 멈추기');
    pp.innerHTML = '<svg viewBox="0 0 14 14" aria-hidden="true">' + (paused ? PLAY : PAUSE) + '</svg>';
  });
  pp.innerHTML = '<svg viewBox="0 0 14 14" aria-hidden="true">' + PAUSE + '</svg>';
  // 화면 밖에 있을 때는 갱신을 쉬어서 여러 요소가 한꺼번에 돌지 않게 합니다
  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (es) { if (!root.isConnected) return io.disconnect(); seen = es[0].isIntersecting; });
    io.observe(root);
  }
  ys = M[0].v.map(Y); paint(ys);
  digits(text(M[0]), !rm); badge(M[0]);
  schedule();
})();
</script>
