<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var btn = root.querySelector('.u004-btn');
  var pct = root.querySelector('.u004-pct');
  var eta = root.querySelector('.u004-eta');
  var live = root.querySelector('.u004-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var raf = 0, timers = [], p = 0, t0 = 0, DUR = rm ? 600 : 3200;
  function later(fn, ms) { timers.push(setTimeout(function () { if (root.isConnected) fn(); }, ms)); }
  function clear() { timers.forEach(clearTimeout); timers = []; cancelAnimationFrame(raf); raf = 0; }
  function set(v) {
    p = v;
    btn.style.setProperty('--p', v.toFixed(4));
    pct.textContent = Math.round(v * 100) + '%';
  }
  // 실제 네트워크처럼 빨라졌다 느려졌다 하는 진행 곡선입니다
  function curve(x) { return Math.min(1, x + Math.sin(x * Math.PI * 3) * 0.045 * (1 - x)); }
  function step(now) {
    if (!root.isConnected) return;
    var x = Math.min(1, (now - t0) / DUR);
    set(curve(x));
    var left = Math.max(0, Math.ceil((1 - x) * DUR / 1000 * 2.2));
    eta.textContent = left > 0 ? '남은 시간 ' + left + '초' : '마무리하는 중';
    if (x < 1) raf = requestAnimationFrame(step);
    else finish();
  }
  function finish() {
    root.dataset.state = 'fold';
    live.textContent = '다운로드 완료';
    later(function () { root.dataset.state = 'done'; }, rm ? 100 : 650);
    later(reset, rm ? 1600 : 3400);
  }
  function reset() { clear(); root.dataset.state = 'idle'; set(0); live.textContent = ''; }
  btn.addEventListener('click', function () {
    var s = root.dataset.state;
    if (s === 'run') { // 받는 중에 누르면 취소하고 처음으로 돌아갑니다
      clear();
      root.dataset.state = 'idle';
      set(0);
      live.textContent = '다운로드를 취소했어요';
      return;
    }
    if (s !== 'idle') { reset(); return; }
    clear();
    set(0);
    root.dataset.state = 'run';
    live.textContent = '다운로드 시작';
    later(function () { t0 = performance.now(); raf = requestAnimationFrame(step); }, rm ? 0 : 260);
  });
})();
</script>
